\section{Conditional certification: finite-sample upper bounds}\label{sec:cert}

Goal: within the model class, give finite-sample upper bounds on $\pL(d)$ at large code distance \emph{directly}, rather than estimating $\alpha$ first and extrapolating.
The proof skeleton has three layers: \emph{(A) the statistical layer} (confidence upper bounds on the parameters from data); \emph{(B) the monotonicity layer} (upper bounds on the parameters give a ``dominating'' model,
and Theorem~\ref{thm:monotone} gives an upper bound on $\err$); \emph{(C) the computational layer} (an explicit upper bound on $\err_d$ for the dominating model).
This section proves the code-capacity case completely (Theorem~\ref{thm:cc-cert}), extends it to code capacity with unflagged burst and chip-scale events (Theorem~\ref{thm:burst-cert}),
gives a separate certificate for the MWPM decoder (\S\ref{sec:mwpm-cert}), states the burst results for any linear decoding structure, i.e.\ at circuit level (\S\ref{sec:spacetime}), and tightens the code-capacity certificate (\S\ref{sec:o7}).

\paragraph{Which decoder.}
All upper bounds in this section are upper bounds for \emph{maximum-likelihood (Bayes-optimal) decoding}, stated as ``there is a decoder (ML with the noise parameters known)
whose logical failure probability does not exceed $U$''. By optimality, this is \emph{not} an upper bound on the failure rate of the suboptimal decoder actually used in an experiment (such as MWPM).
Conversely, the \emph{lower bounds} of this section (the floor, Lemma~\ref{lem:floor}) hold for any decoder.

\subsection{(C) The union (Bhattacharyya) upper bound}

\begin{lemma}[Union bound]\label{lem:union}
Let $(E,F)$ be any noise instance on a linear decoding structure $(V,\sigma,L)$, and let $K=\ker\sigma\setminus\ker L$
(faults that are undetected but flip an observable). Then
\[
\err(E,F)\;\le\;\tfrac12\sum_{G\in K}\BC(P;G),\qquad
\BC(P;G):=\sum_{E,F}\sqrt{P(E,F)\,P(E+G,F)}.
\]
\hfill\st{PROVED}
\end{lemma}
\begin{proof}
Fix an observation $y=(s,f)$, write $P(\ell,y)=\Prob[L=\ell,Y=y]$, and let the ML decision be $\ell^\ast(y)$. Then
$\err=\sum_y\sum_{\ell\ne\ell^\ast}P(\ell,y)$. For $\ell\ne\ell^\ast$ we have $P(\ell,y)\le P(\ell^\ast,y)$, hence
$P(\ell,y)\le\sqrt{P(\ell,y)P(\ell^\ast,y)}$, and so
\[
\err\le\sum_y\sum_{\{\ell,\ell'\}}\sqrt{P(\ell,y)P(\ell',y)}\quad(\text{sum over unordered label pairs}).
\]
Write $c(s,\ell)=\{E:\sigma(E)=s,L(E)=\ell\}$, so that $P(\ell,s,f)=\sum_{A\in c(s,\ell)}P(A,f)$.
From $\sqrt{\sum_ix_i}\le\sum_i\sqrt{x_i}$ we get
\[
\sqrt{P(\ell,s,f)P(\ell',s,f)}\le\sum_{A\in c(s,\ell)}\sum_{B\in c(s,\ell')}\sqrt{P(A,f)P(B,f)}.
\]
Summing over $s$, $\{\ell,\ell'\}$, and $f$, the right side becomes a sum over \emph{unordered} fault pairs $\{A,B\}$ with $\sigma(A)=\sigma(B)$ and $L(A)\ne L(B)$.
Put $G=A+B\in K$; the map $(A,B)\mapsto(A,G)$ is a bijection from ordered pairs onto $V\times K$, and the number of unordered pairs is half of that,
which gives $\tfrac12\sum_{G\in K}\sum_{A,f}\sqrt{P(A,f)P(A+G,f)}$.
\end{proof}

\begin{corollary}[Product form and weight enumerators at code capacity]\label{cor:product}
For $\Nn_{\rm cc}(p,e)$ and a stabilizer code $(n,S)$ we have $K=N(S)\setminus S$, and for $G\in K$
$\BC(P;G)=B(p,e)^{\wt(G)}$, with $B=e+(1-e)\beta_{\rm dep}(p)$ ($\wt$ the Pauli weight). Hence
\[
\err_d(p,e)\le\tfrac12\widetilde W_d\big(B(p,e)\big),\qquad
\widetilde W_d(x)=\sum_{G\in N(S)\setminus S}x^{\wt(G)}=W_{N(S)}(x)-W_S(x),
\]
and by the MacWilliams identity $W_{N(S)}(x)=\dfrac1{|S|}\sum_{g\in S}(1+3x)^{n-\wt(g)}(1-x)^{\wt(g)}$ this can be computed from the weight distribution of the stabilizer group.
\hfill\st{PROVED}
\end{corollary}
\begin{proof}
Single qubit: with flag $f=1$ (probability $e$) the law is uniform and invariant under the shift, contributing $e\cdot1$; with flag $f=0$ the contribution is
$(1-e)\sum_a\sqrt{\Dep_p(a)\Dep_p(a+G_j)}=(1-e)\beta_{\rm dep}(p)$ (the three shifts $G_j\in\{X,Y,Z\}$ are symmetric;
the pairs $(I,G_j),(G_j,I)$ give $2\sqrt{(1-p)p/3}$, and the other two nontrivial Paulis pair with each other and give $2p/3$).
Qubits with $G_j=I$ contribute $1$. Independence gives the product form. The MacWilliams identity is a standard result.
\end{proof}

\noindent For the three $d=3$ codes the inequality $\err\le\tfrac12\widetilde W(B)$ passes in \texttt{theory/checks} (check C5).
For general $d$, $W_S$ can be computed by tensor networks (a 4-dimensional leg per lattice site; the transfer-matrix width grows exponentially with $d$); the feasible range of $d$ is to be benchmarked, and \emph{no commitment is made here}.

\subsection{(A) The statistical layer: confidence upper bounds on the parameters}

\begin{lemma}[Estimating $e$ from the flags]\label{lem:flags}
Among $M$ independent samples there are $nM$ qubit-samples in total, and the flags $F_{ij}$ are i.i.d.\ Bernoulli($e$). Let
$\bar e=\bar e(\delta_1)$ be the one-sided Clopper--Pearson upper limit; then $\Prob[e\le\bar e]\ge1-\delta_1$.\hfill\st{PROVED}
\end{lemma}

\begin{lemma}[Estimating $\lambda$ from syndrome frequencies]\label{lem:lambda}
Fix an $X$-type stabilizer generator $g_0$ of weight $w_0$. Among the samples in which none of the qubits in its support is flagged (say there are $M'$ of them),
the syndrome bit is i.i.d.\ Bernoulli with parameter $\tfrac12(1-\lambda^{w_0})$ (exact when conditioning on the flag pattern).
Let $\overline{P}$ be the Clopper--Pearson upper limit for this parameter (confidence level $1-\delta_2$); then
$\lambda\ge\underline\lambda:=(1-2\overline P)_+^{1/w_0}$, that is, $p\le\bar p:=\tfrac34(1-\underline\lambda)$,
with probability $\ge1-\delta_2$.\hfill\st{PROVED}
\end{lemma}
\begin{proof}
Without flags, every qubit in the support of $g_0$ is depolarized with probability $p$; an $X$-type detector flips if and only if the number of $\{Y,Z\}$-type errors in the support is odd,
and each qubit contributes the eigenvalue $1-\tfrac43p=\lambda$, so $\Pr[\text{bit}=1]=\tfrac12(1-\lambda^{w_0})$.
Conditioned on the flags the samples are independent, hence the count is exactly binomial.
\end{proof}

\begin{theorem}[Finite-sample certification at code capacity]\label{thm:cc-cert}
Let the data be $M$ independent samples (flags and syndromes) of code-capacity noise $\Nn_{\rm cc}(p,e)$, and let $\delta=\delta_1+\delta_2$.
Let $\bar e,\bar p$ be as in Lemmas~\ref{lem:flags} and~\ref{lem:lambda}. Then with probability at least $1-\delta$, simultaneously for \emph{all} $d$,
\[
\err_d(p,e)\;\le\;\err_d(\bar p,\bar e)\;\le\;\tfrac12\widetilde W_d\big(B(\bar p,\bar e)\big).
\]
\hfill\st{PROVED}
\end{theorem}
\begin{proof}
On the event $\{e\le\bar e\}\cap\{p\le\bar p\}$, Theorem~\ref{thm:preorder} (case $e'\ge e$, $p'\ge p$) shows that
$\Nn_{\rm cc}(\bar p,\bar e)$ is obtained from $\Nn_{\rm cc}(p,e)$ by a local free operation (superpose erasure, then superpose depolarizing noise),
so Theorem~\ref{thm:monotone} gives the first inequality; the second is Corollary~\ref{cor:product}.
The probability of this event is $\ge1-\delta_1-\delta_2$ by the union bound.
\end{proof}

This is a certificate that \emph{does not depend on any $\pL$ data}: flag and syndrome statistics suffice to give upper bounds for all $d$. The price is the looseness of the union bound: at the distances used in practice it behaves as if it had a threshold far below the true ML threshold, and because it counts every element of every logical class it grows without bound as $d\to\infty$ at any fixed noise (Proposition~\ref{prop:lower}(b); the crossover is at $d$ of order $10^3$--$10^5$ for the parameters of interest). For large $d$ use Proposition~\ref{prop:ml-peierls} instead. Replacing the union bound by a tighter, equally monotone bound is O7.

\noindent\emph{Relation to current practice.} Published suppression factors are fits with statistical error bars (for example $\Lambda=2.14\pm0.02$ in~\cite{Google24}; see also~\cite{Google22}), and extrapolating them to larger $d$ gives an estimate, not a bound. Rare-event sampling~\cite{BV} reaches small failure probabilities at fixed parameters, again as an estimate. Takou et al.~\cite{Takou} estimate detector-error-model probabilities from the syndrome data of surface-code memory experiments (Google's Willow data and IBM data) and use them as decoder priors to estimate and reduce logical error probabilities. Estimating error rates from syndrome statistics is older: Spitz, Tarasinski, Beenakker and O'Brien~\cite{Spitz} determine the matching weights from correlators of pairs of detection events, without an error model, and adapt them to a time-dependent environment; the $p_{ij}$ method of~\cite{Google21} estimates the probability of each pair-correlated error mechanism from the two-point correlations of detection events. We did not find finite-sample upper bounds on $\pL$ valid for all $d$, such as Theorems~\ref{thm:cc-cert}, \ref{thm:burst-cert} and~\ref{thm:st-cert}, in this literature.

\subsection{The case with local bursts and chip-scale events}

Correlated noise and bursts have a long history. Threshold theorems for correlated noise include Terhal and Burkard~\cite{TerhalBurkard} (local non-Markovian noise) and Aharonov, Kitaev and Preskill~\cite{AKP} (correlations decaying faster than $1/r^D$). Tan, Pattison, McEwen and Preskill~\cite{TPMP} determine by Monte Carlo simulation (circuit-level depolarizing noise, a burst lasting one syndrome cycle and uniform across the block) the region of burst and background error rates in which the memory time grows without bound with the block size; Suzuki et al.~\cite{Q3DE} propose detecting multi-bit bursts from syndrome values and mitigating them by code deformation.
The results below are certified, finite-sample statements for the model class of Definition~\ref{def:events}.

Let the model class be that of Definition~\ref{def:events}. Let $q_c=1-e^{-r_c\,T_{\rm tot}}$ be the probability of at least one chip-scale event within the total time $T_{\rm tot}$.

\begin{lemma}[Floor from chip-scale events]\label{lem:floor}
Let ${\err}^{(0)}$ be the $\err$ of the same noise with the chip-scale layer removed. Then
\[
\err\;\le\;q_c+(1-q_c)\,{\err}^{(0)}.
\]
If in addition \textbf{(K)} holds: given a chip event, the observable $L(E)\in\F^q$ is uniform and independent of $Y=(\sigma(E),F)$, then
\[
\err\;\ge\;q_c\,(1-2^{-q}).
\]
So the ``floor'' $\err\sim q_c\propto r_cT_{\rm tot}$ is independent of $d$; the upper bound does not need (K), and the lower bound holds equally for the failure rate of any decoder.
\hfill\st{PROVED}
\end{lemma}
\begin{proof}
Upper bound: process the full problem with the optimal ML decoder $\hat g_0$ for the problem without the chip layer. Let $C$ be the event ``at least one chip event''.
$\Prob[\hat g_0\text{ fails}]\le\Prob[C]+\Prob[C^c]\,{\err}^{(0)}$, because on $C^c$ the data follow the noise without the chip layer (the layers are independent).
$\err$ does not exceed the failure rate of any decoder.

Lower bound: consider a genie decoder that knows $C$; its failure rate does not exceed $\err$ (the (F1) case of Theorem~\ref{thm:monotone}: the real observation is a coarse-graining of the genie's observation).
On $C$, by (K) any guess fails with probability $1-2^{-q}$.
(K) holds for ``whole-chip completely depolarizing events'': a uniform $E_{\rm chip}$ makes $(\sigma,L)(E)$ uniform on $\F^{n-k}\times\F^{2k}$ (surjectivity),
so $L$ is independent of $\sigma$; the flags are independent of $E_{\rm chip}$.
\end{proof}
\noindent\emph{Status.} As a qualitative statement Lemma~\ref{lem:floor} is established: McEwen et al.~\cite{McEwen} resolved chip-wide error bursts from high-energy impacts in a superconducting processor; a single high-energy event set a floor of $1.7\times10^{-6}$ logical errors per round for a distance-$25$ repetition code in~\cite{Google22}; rare correlated events, about once per hour, limit the repetition codes of~\cite{Google24}; and chip-level catastrophic errors motivate error correction across chips~\cite{XuChip}. The lemma only makes the floor explicit, with constants, in the present model class.

\begin{lemma}[Union bound for burst mixtures]\label{lem:burst-union}
Let $E=E_{\rm bg}+E_{\rm burst}$, and suppose that conditioned on the burst configuration $\mathcal C$ the noise is a product measure over qubits, with configuration law $\pi$. Then
\[
\BC(P;G)\;\le\;\sum_{\mathcal C,\mathcal C'}\sqrt{\pi(\mathcal C)\pi(\mathcal C')}\;\prod_{j}\beta_j(\mathcal C,\mathcal C';G_j),
\]
where $\beta_j(\mathcal C,\mathcal C';G_j)=\sum_{a,f}\sqrt{P_{\mathcal C,j}(a,f)P_{\mathcal C',j}(a+G_j,f)}$.
\hfill\st{PROVED}
\end{lemma}
\begin{proof}
$P=\sum_{\mathcal C}\pi(\mathcal C)P_{\mathcal C}$. From $\sqrt{\sum x_i}\le\sum\sqrt{x_i}$,
\[
\sqrt{P(E,F)P(E+G,F)}\le\sum_{\mathcal C,\mathcal C'}\sqrt{\pi(\mathcal C)\pi(\mathcal C')}\sqrt{P_{\mathcal C}(E,F)P_{\mathcal C'}(E+G,F)}.
\]
Sum over $(E,F)$ and use the product measure.
\end{proof}
\noindent Note that one cannot simply ``condition on $\mathcal C$ and treat $\mathcal C$ as known'': that yields the failure rate of a decoder that knows the configuration,
which is \emph{lower} than $\err$ (the genie is stronger), not an upper bound. Lemma~\ref{lem:burst-union} is the correct treatment of an unknown configuration, but its right side sums over two independent burst configurations simultaneously
and is therefore loose.

\subsubsection*{A single-sum union bound for burst mixtures (O3)}

The double sum of Lemma~\ref{lem:burst-union} can be avoided. The key is a tilted form of the Bhattacharyya coefficient.

\begin{lemma}[Tilted Bhattacharyya bound]\label{lem:tilted}
Let $P$ be any law of $(E,F)$, $G\in V$, and $Q>0$ any positive function of $(E,F)$ (not necessarily normalized). Then
\[
\BC(P;G)\;\le\;\E_P\Big[\sqrt{Q(E+G,F)/Q(E,F)}\Big],
\]
with equality for $Q=P$ (where $P>0$).\hfill\st{PROVED}
\end{lemma}
\begin{proof}
Write $x=(E,F)$, $x+G=(E+G,F)$ and $h(x)=\sqrt{Q(x+G)/Q(x)}$. By the AM--GM inequality, $\sqrt{P(x)P(x+G)}\le\tfrac12P(x)h(x)+\tfrac12P(x+G)/h(x)$.
Sum over $x$ and substitute $y=x+G$ in the second term; since $G+G=0$, $1/h(y+G)=h(y)$, so both terms equal $\tfrac12\E_P[h]$.
\end{proof}

\begin{definition}[Code-capacity bursts]\label{def:cc-bursts}
On a stabilizer code with $n$ qubits, the noise $\Nn_{\rm cc}^{\rm burst}$ consists of: background $\Nn_{\rm cc}(p,e)$; for each region $B_z$ of a finite family of subsets of qubits, an independent Poisson number of events of intensity $r_z$, each of which applies independent depolarizing noise $\Dep_{p_b}$ to every qubit of $B_z$, with a strength $p_b\in[0,\tfrac34]$ drawn independently from an arbitrary (unknown) law; and, optionally, a chip-scale layer as in Lemma~\ref{lem:floor}. Bursts carry no flags and faults add in $\F$. Put
\[
V=\max_z|B_z|,\qquad \rho_z=1-e^{-r_z},\qquad \bar\rho=\max_j\sum_{z:\,j\in B_z}\rho_z
\]
(largest region and largest expected number of hit regions covering a qubit).
\end{definition}

\begin{theorem}[Product-form union bound for burst mixtures]\label{thm:burst-product}
For $\Nn_{\rm cc}^{\rm burst}$ without the chip layer, every $t\in(0,\tfrac34]$, $\tau=t/(3(1-t))$ and
\[
m(s)=e+(1-e)\Big[(1-s)\sqrt\tau+\frac{s}{3\sqrt\tau}+\frac{2s}3\Big],\qquad \kappa=\frac{m(3/4)}{m(p)},\qquad
B_{\rm burst}=m(p)\,\exp\!\Big(\bar\rho\,\frac{\kappa^V-1}V\Big),
\]
we have $\BC(P;G)\le B_{\rm burst}^{\wt(G)}$ for every $G$, and therefore
\[
\err_d\;\le\;\tfrac12\,\widetilde W_d\big(B_{\rm burst}\big).
\]
For $t=p$, $m(p)=B(p,e)$, so the bound reduces to Corollary~\ref{cor:product} when $\bar\rho=0$. The bound depends on the burst layer only through $V$ and $\bar\rho$: it holds uniformly over all strength laws.\hfill\st{PROVED}
\end{theorem}
\begin{proof}
Apply Lemma~\ref{lem:tilted} with $Q=\prod_jQ_j$, $Q_j(a,0)=\Dep_t(a)$, $Q_j(a,1)=\tfrac14$. The factor $g_j=\sqrt{Q_j(a+G_j,f)/Q_j(a,f)}$ is $1$ if $G_j=I$ or $f=1$; for $f=0$ and $G_j\ne I$ it is $\sqrt\tau$ at $a=I$, $1/\sqrt\tau$ at $a=G_j$, and $1$ at the other two Paulis.
Condition on the burst configuration $\mathcal C$ (event counts and strengths). The qubits are then independent; an unflagged qubit $j$ carries $\Dep_{s_j}$ with $s_j=p$ if no event covers it and $s_j\in[p,\tfrac34]$ otherwise (depolarizing rates compose within $[p,\tfrac34]$), and an erased qubit is uniform. Hence $\E[g_j\mid\mathcal C]=m(s_j)$ for $j\in\supp G$.
$m$ is affine with slope $(1-e)(1+3u)(1-u)/(3u)\ge0$, $u=\sqrt\tau\le1$, so $m(s_j)\le m(p)$ if $j$ is uncovered and $m(s_j)\le m(\tfrac34)$ otherwise. With $N_G$ the number of covered qubits of $\supp G$ and $c_z=\mathbf 1[\text{at least one event in }B_z]$ (independent Bernoulli($\rho_z$)),
\[
\E_P\Big[\prod_jg_j\Big]\le m(p)^{\wt G}\,\E\,\kappa^{N_G}\le m(p)^{\wt G}\prod_z\big(1+\rho_z(\kappa^{a_z}-1)\big)\le m(p)^{\wt G}\exp\Big(\sum_z\rho_z(\kappa^{a_z}-1)\Big),
\]
where $a_z=|B_z\cap\supp G|\le V$ and $N_G\le\sum_zc_za_z$ ($\kappa\ge1$). By convexity, $\kappa^{a}-1\le\tfrac aV(\kappa^V-1)$ for $0\le a\le V$, and $\sum_z\rho_za_z=\sum_{j\in\supp G}\sum_{z\ni j}\rho_z\le\bar\rho\,\wt G$.
This gives $\BC(P;G)\le B_{\rm burst}^{\wt G}$; Lemma~\ref{lem:union} and $K=N(S)\setminus S$ give the second claim. For $t=p$, $m(p)=2\sqrt{p(1-p)/3}+2p/3$ on unflagged qubits, i.e.\ $m(p)=B(p,e)$.
\end{proof}

\noindent The right-hand side is the weight enumerator of the code evaluated at one number, exactly as without bursts, so at each $d$ the bursts act as a larger single-qubit coefficient. \emph{Correction:} an earlier version concluded from this that bursts produce no $d$-independent floor. That does not follow: $\widetilde W_d(x)$ grows without bound as $d\to\infty$ for every $x>0$ (Proposition~\ref{prop:lower}(b)), so this bound says nothing about $d\to\infty$. The asymptotic statement is proved by the Peierls route instead (Theorem~\ref{thm:peierls-bursts}). The price in both is the factor $\kappa^V$: a hit region can conspire with the background on up to $V$ qubits.
The free parameter $t$ can be optimized after the data are seen (the inequality holds for every $t$). Check K3 of \texttt{checks/completion\_checks.py} verifies $\BC\le\E_P[h]\le B_{\rm burst}^{\wt G}$ for every logical $G$, and $\err\le\tfrac12\sum_G\BC\le\tfrac12\widetilde W(B_{\rm burst})$, with exact ML on $[[5,1,3]]$ (with erasure) and on the $d=3$ rotated surface code.
The same argument applies to any structure in which, given the burst configuration, the fault law is a product over fault locations (space-time locations included); only the single-location factor $m$ changes. This extension is not written out here.

\subsubsection*{The shape question (O1) has a negative answer}

\begin{proposition}[$\err$ is not monotone in the burst strength]\label{prob:O1}
On the $d=3$ rotated surface code take background $p=\tfrac1{200}$, $e=0$, and one region $B=\{(0,1),(1,0),(1,2),(2,1)\}$ that is hit with probability $\rho=\tfrac1{20}$, in which case $\Dep_{p_b}$ is added on its qubits (unflagged). Then
\[
\err(p_b=\tfrac12)=4.8396\times10^{-3}\;>\;\err(p_b=\tfrac34)=3.5580\times10^{-3}
\]
(exact rational arithmetic; the values are rounded). So for unflagged bursts $\err$ is not monotone in the shape parameter $p_b$.\hfill\st{PROVED} (exact computation)
\end{proposition}
\noindent The computation is check K4 of \texttt{checks/completion\_checks.py}. The same drop occurs in the Poisson form of Definition~\ref{def:cc-bursts} (intensity $0.05$: $4.72\times10^{-3}\to3.48\times10^{-3}$, floating point), and a random search finds smaller examples on $n=3$ qubits.
Monotonicity also fails in the size of the region and in the duration (Observations~\ref{obs:O1-region}, \ref{obs:O1-duration}). Since monotonicity fails, certification must not rely on a shape upper bound. Theorem~\ref{thm:burst-product} does not: it is uniform in the strength, and it uses only the region size $V$ and the rates.

\subsubsection*{Parameter limits under contamination (O2)}

\begin{lemma}[Contamination-robust confidence limits]\label{lem:robust}
Let the data be $M$ independent and identically distributed shots of $\Nn_{\rm cc}^{\rm burst}$ (with the chip layer), where flags come only from erasures.
(a) Lemma~\ref{lem:flags} holds unchanged: $\Prob[e\le\bar e]\ge1-\delta_1$.
(b) Lemma~\ref{lem:lambda} remains valid, and becomes conservative: the syndrome bit of $g_0$ on shots without flags in its support is Bernoulli with parameter $P\ge\tfrac12(1-\lambda^{w_0})$, so $\Prob[p\le\bar p]\ge1-\delta_2$ with the same $\bar p$.
(c) Let $T$ be a binary statistic of a shot and $\mathcal Z$ a set of regions with $\Prob[T=1\mid\text{at least one event in }\mathcal Z]\ge\eta>0$ (a detection efficiency). With $\bar T$ the Clopper--Pearson upper limit for $\Prob[T=1]$, $\Prob\big[\sum_{z\in\mathcal Z}r_z\le-\ln(1-\bar T/\eta)\big]\ge1-\delta_3$ (vacuous if $\bar T\ge\eta$).
\hfill\st{PROVED}
\end{lemma}
\begin{proof}
(a) The flags are independent of the other layers. (b) The bit is the parity of the background contribution plus an independent contamination term, so $\E(-1)^{\rm bit}=\lambda^{w_0}\chi$ with $|\chi|\le1$ and $\lambda^{w_0}\ge0$. Hence $1-2P\le\lambda^{w_0}$, and on the event $P\le\overline P$ we get $\lambda\ge(1-2\overline P)_+^{1/w_0}$. (c) $\Prob[T=1]\ge\eta\,(1-e^{-\sum_{z\in\mathcal Z}r_z})$; on the event $\Prob[T=1]\le\bar T$ this rearranges to the stated limit.
\end{proof}

\begin{proposition}[No uniformly valid upper limit on the burst rate without a strength floor]\label{prop:no-rate-limit}
Suppose a procedure computes $\bar r$ from $M$ shots and has $\Prob_\theta[r_z\le\bar r]\ge1-\delta$ for every model $\theta$ of a class containing, for one region $z$, all models $\theta_{R,\epsilon}$ (rate $R$, strength $\epsilon$) with $R>0$ and $\epsilon\in(0,\epsilon_0]$. Then under the burst-free model $\theta_0$, $\Prob_{\theta_0}[\bar r=\infty]\ge1-\delta$.\hfill\st{PROVED}
\end{proposition}
\begin{proof}
The single-shot law of $\theta_{R,\epsilon}$ is continuous in $\epsilon$ and equals that of $\theta_0$ at $\epsilon=0$, so $\mathrm{TV}(P^{\otimes M}_{\theta_{R,\epsilon}},P^{\otimes M}_{\theta_0})\to0$ as $\epsilon\to0$. Hence $\Prob_{\theta_0}[\bar r\ge R]\ge\Prob_{\theta_{R,\epsilon}}[\bar r\ge R]-\mathrm{TV}\ge1-\delta-o(1)$ for every $R$; let $R\to\infty$.
\end{proof}

\noindent So O2 is settled for $p$ and $e$ (shots i.i.d.; for multi-round data use windows separated by gaps $\ge T_b$, whose event counts are independent), while an upper limit on the burst rates needs an assumption that makes bursts detectable, such as a floor on the strength that yields the efficiency $\eta$ in (c). A certificate that needs no such floor, because weak bursts are absorbed into the background estimate, is Theorem~\ref{thm:weak-bursts}. The circuit-level version of (b) is Lemma~\ref{lem:st-robust}.

\subsubsection*{The certification theorem with bursts and chip events}

\begin{theorem}[Conditional certification with bursts and chip events, code capacity]\label{thm:burst-cert}
Let the true model be $\Nn_{\rm cc}^{\rm burst}$ with parameters $\theta=(p,e,(r_z),r_c)$, known regions $(B_z)$, and arbitrary unknown burst strength law and chip fault law. Let $q_c=1-e^{-r_c}$ be the probability of a chip event in a shot.
Suppose the confidence upper limits satisfy $\Prob[p\le\bar p,\ e\le\bar e,\ r_z\le\bar r_z\ \forall z,\ r_c\le\bar r_c]\ge1-\delta$ (for example from Lemma~\ref{lem:robust}). Then with probability $\ge1-\delta$, for all $d$ and all $t\in(0,\tfrac34]$,
\[
\err_d(\theta)\;\le\;\bar q_c+(1-\bar q_c)\,\min\Big\{1,\ \tfrac12\widetilde W_d\big(B_{\rm burst}(\bar p,\bar e,\bar\rho,V;t)\big)\Big\},
\qquad\bar q_c=1-e^{-\bar r_c},
\]
where $\bar\rho$ is computed from the $\bar r_z$.\hfill\st{PROVED}
\end{theorem}
\begin{proof}
On the confidence event, the model $\bar\theta$ with the same strength and chip laws and parameters $(\bar p,\bar e,(\bar r_z),\bar r_c)$ is obtained from $\theta$ by superposing independent, classically samplable layers. These are depolarizing noise on unflagged qubits and erasure (Lemma~\ref{lem:basic-ops}; erasure overwrites the other layers), and further Poisson events of rates $\bar r_z-r_z$ and $\bar r_c-r_c$ (superposition theorem).
Hence $\err(\theta)\le\err(\bar\theta)$ (Proposition~\ref{prop:rate-mono}). Lemma~\ref{lem:floor} gives $\err(\bar\theta)\le\bar q_c+(1-\bar q_c){\err}^{(0)}(\bar\theta)$, and Theorem~\ref{thm:burst-product} bounds ${\err}^{(0)}(\bar\theta)$ (and trivially ${\err}^{(0)}\le1$).
\end{proof}

\noindent \emph{Coverage check} (L4 of \texttt{checks/second\_pass\_checks.py}): at $d=3$ with Poisson bursts on one plaquette, $\bar e$ from flags, $\bar p$ from a weight-$2$ check off the region, and $\bar r$ from the region's own check with the efficiency $\eta=\tfrac12(1-\lambda(0.2)^4)$ of Lemma~\ref{lem:robust}(c), the confidence event has coverage $0.95$--$0.99$ ($300$ runs per point, target $0.95$) and the certificate exceeds the exact $\err$ on every covering run; but it is vacuous there (median $80$--$100$): the factor $\kappa^V$ with $\bar\rho\approx\bar r$ (dominated by background firings of the detector) dominates at $d=3$.

\noindent Compared with the earlier target statement: no monotonicity in the shape is needed (Proposition~\ref{prob:O1} shows it fails), and the double configuration sum is replaced by the product bound. The space-time (circuit-level) version is Theorem~\ref{thm:st-cert}; what it still lacks for a computed number is the location-weight enumerator of the detector error model.

\paragraph{Division of numerical checks (see \S\ref{sec:map}).}
The circuit-level stim + MWPM data of P2 can check: (i) the floor $\err\ge q_c(1-2^{-q})$ and its scaling $\propto r_cT_{\rm tot}$
(the lower bound holds for any decoder, so the MWPM failure rate must not fall below it); (ii) unbiasedness of the parameter estimators (milestone M3); and
(iii) that bursts reduce the exponent rather than create a floor. The \emph{upper-bound} parts concern ML, and MWPM data cannot falsify them;
they must be checked with exact ML at small $d$ (\texttt{lcd.decoders.ExactTNMLDecoder}) or with tensor-network ML for code capacity with burst clusters (M1).

\subsection{Certification for the MWPM decoder at code capacity (O5)}\label{sec:mwpm-cert}

The bounds above concern ML decoding. For minimum-weight matching a Peierls-type argument \cite{DKLP} gives an upper bound directly, with no monotonicity property of the decoder needed. Rigorous threshold theorems for circuit-level noise with explicit constants are also known for concatenated codes, by a different, inductive argument: Aliferis, Gottesman and Preskill~\cite{AGP} prove one for concatenated distance-3 codes, including noise correlated in space and time, with a computer-assisted lower bound on the threshold. For matching on the surface code the analogue is the Peierls route of~\cite{DKLP,Fowler}, taken to circuit level in \S\ref{sec:circuit-peierls}.

\begin{theorem}[Peierls bound for MWPM with erasures]\label{thm:mwpm-peierls}
Consider the rotated surface code of odd distance $d$ (conventions of \texttt{lcd.codes.RotatedSurfaceCode}) under $\Nn_{\rm cc}(p,e)$. The decoder corrects the $X$ part and the $Z$ part independently, each by a correction of minimum weight with the observed syndrome, where the weight counts the non-erased qubits (erased qubits have weight $0$; ties are broken arbitrarily). At $e=0$ this is the decoder \texttt{MWPMCodeCapacity}. Let $q=2p/3$ and
\[
B_M(p,e)=e+(1-e)\,2\sqrt{q(1-q)}.
\]
If $3B_M<1$, the logical failure probability of this decoder satisfies
\[
\pL^{\rm MWPM}(p,e)\;\le\;2\sum_{\ell\ge d}d\,3^{\ell-1}B_M^{\ell}\;=\;\frac{2d\,B_M\,(3B_M)^{d-1}}{1-3B_M}.
\]
\hfill\st{PROVED}
\end{theorem}
\begin{proof}
We treat the $X$ part; the $Z$ part is the same with the roles of rows and columns exchanged, and the union bound gives the factor $2$.
\emph{Matching graph.} Every qubit lies in one or two $Z$ checks (its four neighbouring plaquettes alternate in type, and a qubit in no $Z$ check would be an undetectable weight-$1$ error). Form the graph whose vertices are the $Z$ checks and a boundary vertex $\beta$, in which qubit $j$ is an edge joining its two $Z$ checks, or its $Z$ check and $\beta$.
Every $Z$ check has weight $\le4$, so every vertex other than $\beta$ has degree $\le4$. The qubits of row $0$ lie in exactly one $Z$ check each (check K1), so their edges end at $\beta$.

\emph{Failure produces a heavy cycle.} Let $E$ be the $X$ part of the error and $C$ the correction. $\Gamma=E\oplus C$ has zero syndrome, so every check vertex has even degree in $\Gamma$, hence so does $\beta$, and $\Gamma$ is an edge-disjoint union of simple cycles.
A failure means that $\Gamma$ has odd overlap with the $Z$ logical on row $0$, so some simple cycle $\gamma\subseteq\Gamma$ has odd overlap with row $0$. Then $\gamma$ contains a row-$0$ edge, hence passes through $\beta$, and it is an $X$ logical operator, so $|\gamma|\ge d$.
$C\oplus\gamma$ has the same syndrome as $C$, so minimality gives $w(\gamma\cap C)\le w(\gamma\setminus C)$, where $w$ counts non-erased qubits. Since $\gamma\subseteq E\oplus C$ we have $\gamma\setminus C=\gamma\cap E$, so at least half of the non-erased edges of $\gamma$ carry an $X$ error.

\emph{Probability for a fixed cycle.} The $X$ components of non-erased qubits are independent Bernoulli($q$), and erased qubits impose no constraint. For $t\ge1$ the indicator that at least half of the $k$ non-erased edges are in error is at most $\prod_{\text{non-erased}}t^{x_i-1/2}$. Taking the expectation edge by edge and choosing $t=(1-q)/q\ge1$ ($q\le\tfrac12$) bounds the probability for a cycle of length $\ell$ by $\big(e+(1-e)2\sqrt{q(1-q)}\big)^\ell=B_M^\ell$.

\emph{Counting.} A simple cycle of length $\ell$ through $\beta$ that uses a row-$0$ edge is traced by leaving $\beta$ along a row-$0$ edge ($d$ choices) and then choosing each of the remaining $\ell-1$ edges at a vertex of degree $\le4$, other than the edge just used ($\le3$ choices). There are at most $d\,3^{\ell-1}$ of them. Summing over $\ell\ge d$ gives the claim.
\end{proof}

\begin{corollary}[Finite-sample MWPM certificate]\label{cor:mwpm-cert}
With $\bar p,\bar e$ from Lemmas~\ref{lem:flags} and~\ref{lem:lambda} (or Lemma~\ref{lem:robust}) at total level $\delta$, with probability $\ge1-\delta$, simultaneously for all $d$ with $3B_M(\bar p,\bar e)<1$,
\[
\pL^{\rm MWPM}(p,e)\;\le\;\frac{2d\,B_M(\bar p,\bar e)\,\big(3B_M(\bar p,\bar e)\big)^{d-1}}{1-3B_M(\bar p,\bar e)}.
\]
\hfill\st{PROVED}
\end{corollary}
\begin{proof}
$B_M$ is nondecreasing in $p\in[0,\tfrac34]$ (as $q\le\tfrac12$) and in $e$ (as $2\sqrt{q(1-q)}\le1$), and the right side of Theorem~\ref{thm:mwpm-peierls} is increasing in $B_M$ on $3B_M<1$. Apply it on the confidence event.
\end{proof}

\noindent The argument is that of Dennis, Kitaev, Landahl and Preskill~\cite{DKLP} (counting self-avoiding paths that carry at least half of their weight in errors); Kovalev and Pryadko~\cite{KovalevPryadko} show that every family of low-density parity-check codes whose distance grows as a power of the length corrects all errors with certainty if the error rate is small enough. What is added here is the treatment of erasures as weight-$0$ edges and the closed-form constants.
This answers O5 for code-capacity MWPM. The bound is rigorous but crude: its range of validity $3B_M<1$ is $p<4.29\%$ at $e=0$ and $e<\tfrac13$ at $p=0$, far below the MWPM thresholds. Replacing $3$ by self-avoiding-walk counts improves the range to about $4.9\%$ (Proposition~\ref{prop:saw}).
Because it is a bound for a fixed decoder, it does not conflict with the rule that MWPM data cannot falsify ML bounds: it is falsified only by MWPM failure rates above it. A comparison with simulated MWPM failure rates is listed in \S\ref{sec:map}.

\subsection{Space-time version: bursts and parameter limits for any linear decoding structure}\label{sec:spacetime}

The code-capacity results above use only a product structure over fault locations. This subsection states them for an arbitrary linear decoding structure, in particular for the detector error model of a syndrome-extraction circuit (phenomenological or circuit-level noise), which answers the open parts ``space-time version'' of O2 and O3.

\begin{definition}[Local depolarizing noise on a structure]\label{def:local-dep}
Let $(V,\sigma,L)$ be a linear decoding structure with $V=\bigoplus_{\ell\in\mathcal L}V_\ell$ (fault locations $\ell$; $V_\ell\cong\F^{k_\ell}$, $K_\ell=|V_\ell|\ge2$). The \emph{local depolarizing law} of rate $s\in[0,\tfrac{K-1}K]$ on $V_\ell$ is $\Dep^{(K)}_s=(1-s)\delta_0+\tfrac s{K-1}\mathbf 1_{V_\ell\setminus0}$; $\Dep^{(K)}_{(K-1)/K}$ is uniform.
The class $\Nn_{\rm st}$: background faults independent across locations, location $\ell$ flagged with probability $e_\ell$ (then uniform) and otherwise $\Dep^{(K_\ell)}_{p_\ell}$; burst regions $z\subseteq\mathcal L$ with independent activations, $\Prob[N_z\ge1]\le\rho_z$; each activation adds, independently at every $\ell\in z$, a fault with law $(1-\theta)\delta_0+\theta\,\mathbf 1_{V_\ell}/K_\ell$, $\theta\in[0,1]$ arbitrary (it may differ between events and be random, independently of everything else); faults add in $\F$; bursts carry no flags.
Put $V_{\max}=\max_z|z|$, $\bar\rho=\max_\ell\sum_{z\ni\ell}\rho_z$, and $|G|_{\mathcal L}=\#\{\ell:G_\ell\ne0\}$ (location weight).
\end{definition}
\noindent Examples: code capacity ($\mathcal L$ = qubits, $K=4$, $\theta=\tfrac43p_b$); single-qubit and two-qubit gate locations ($K=4,16$); and a stim detector error model, whose independent error mechanisms are locations with $K=2$ (a mechanism flips with probability $p_\ell\le\tfrac12$, and a burst raises it within $[p_\ell,\tfrac12]$). A local depolarizing law composed with $(1-\theta)\delta_0+\theta\,\mathrm{unif}$ is again local depolarizing, with rate $(1-\theta)s+\theta\tfrac{K-1}K\in[s,\tfrac{K-1}K]$.

\begin{theorem}[Space-time burst union bound]\label{thm:st-burst}
For $\Nn_{\rm st}$, every $t\in(0,\tfrac12]$ and every $G\in K=\ker\sigma\setminus\ker L$,
\[
\BC(P;G)\le\prod_{\ell:\,G_\ell\ne0}m_\ell(p_\ell)\cdot\exp\Big(\sum_z\rho_z\Big(\prod_{\ell\in z,\,G_\ell\ne0}\kappa_\ell-1\Big)\Big)\le\bar B^{\,|G|_{\mathcal L}},\qquad
\bar B=\bar m\exp\Big(\bar\rho\,\frac{\bar\kappa^{V_{\max}}-1}{V_{\max}}\Big),
\]
where, with $\tau_\ell=t/((K_\ell-1)(1-t))$,
\[
m_\ell(s)=e_\ell+(1-e_\ell)\Big[(1-s)\sqrt{\tau_\ell}+\frac{s}{(K_\ell-1)\sqrt{\tau_\ell}}+\frac{s(K_\ell-2)}{K_\ell-1}\Big],\quad\kappa_\ell=\frac{m_\ell(\tfrac{K_\ell-1}{K_\ell})}{m_\ell(p_\ell)}\ge1,
\]
$\bar m=\max_\ell m_\ell(p_\ell)$, $\bar\kappa=\max_\ell\kappa_\ell$. Hence $\err\le\tfrac12\widetilde W^{\mathcal L}(\bar B)$ with the location-weight enumerator $\widetilde W^{\mathcal L}(x)=\sum_{G\in K}x^{|G|_{\mathcal L}}$, uniformly over all burst strengths. For $K=4$ and a common rate this is Theorem~\ref{thm:burst-product}; at $t=p_\ell$, $\bar\rho=0$, $m_\ell(p_\ell)$ is the Bhattacharyya coefficient of location $\ell$.\hfill\st{PROVED}
\end{theorem}
\begin{proof}
The proof of Theorem~\ref{thm:burst-product} goes through location by location. Apply Lemma~\ref{lem:tilted} with $Q=\prod_\ell Q_\ell$, $Q_\ell(a,1)=1$, $Q_\ell(0,0)=1$, $Q_\ell(a,0)=\tau_\ell$ for $a\ne0$. For $G_\ell\ne0$ the factor $\sqrt{Q_\ell(a+G_\ell,f)/Q_\ell(a,f)}$ is $1$ if $f=1$, and for $f=0$ it is $\sqrt{\tau_\ell}$ at $a=0$, $1/\sqrt{\tau_\ell}$ at $a=G_\ell$ and $1$ at the other $K_\ell-2$ points; its expectation under $\Dep^{(K_\ell)}_s$ (with the flag) is $m_\ell(s)$.
Given the activations, location $\ell$ carries $\Dep^{(K_\ell)}_{s_\ell}$ with $s_\ell=p_\ell$ if no active region covers it and $s_\ell\in[p_\ell,\tfrac{K_\ell-1}{K_\ell}]$ otherwise. With $u=\sqrt{\tau_\ell}$ the slope of $m_\ell$ is $(1-e_\ell)\frac{(1-u)(1+(K_\ell-1)u)}{(K_\ell-1)u}\ge0$, because $u\le1$ for $t\le\tfrac12\le\tfrac{K_\ell-1}{K_\ell}$. The rest (independent regions, $\mathbf 1[\text{covered}]\le\sum_{z\ni\ell}\mathbf 1[N_z\ge1]$, $1+x\le e^x$, convexity of $k\mapsto\bar\kappa^k$) is Steps~3--4 of Theorem~\ref{thm:burst-product} with qubits replaced by locations.
\end{proof}

\begin{lemma}[Parameter limits from one detector, space-time]\label{lem:st-robust}
In $\Nn_{\rm st}$ (plus a chip layer independent of everything), let $D\in V^\ast$ be a detector (it fires iff $\langle D,E\rangle=1$) and suppose no location in $\supp D=\{\ell:D_\ell\ne0\}$ is flagged. Then $\Prob[\text{$D$ fires}]=\tfrac12\big(1-\chi\prod_{\ell\in\supp D}\lambda_\ell\big)$ with $\lambda_\ell=1-\tfrac{K_\ell}{K_\ell-1}p_\ell\in[0,1]$ and $\chi\in[-1,1]$ the parity expectation of the event layers. Hence $\Prob[\text{$D$ fires}]\ge\tfrac12(1-\prod_\ell\lambda_\ell)$, and an upper confidence limit $\bar P$ for the firing probability gives $\prod_{\ell\in\supp D}\lambda_\ell\ge1-2\bar P$, which for a one-parameter family $p_\ell=\varphi_\ell(p)$ with nondecreasing $\varphi_\ell$ is an upper limit $\bar p$. Samples must be independent and identically distributed: use one detector per window, with windows whose location sets are disjoint and which are separated in time by more than the event durations $T_b,\tau_c$.\hfill\st{PROVED}
\end{lemma}
\begin{proof}
$\langle D,E\rangle=\sum_\ell\langle D_\ell,E_{\ell}\rangle$ over independent locations plus the independent event contribution, so $\E(-1)^{\langle D,E\rangle}=\chi\prod_\ell\E(-1)^{\langle D_\ell,E_\ell\rangle}$. For $D_\ell\ne0$, $\sum_{a\ne0}(-1)^{\langle D_\ell,a\rangle}=-1$, so $\E_{\Dep_{p}}(-1)^{\langle D_\ell,a\rangle}=(1-p)-\tfrac p{K-1}=\lambda_\ell$. The rest is Lemma~\ref{lem:robust}(b). Windows with disjoint location sets have independent background faults; Poisson events in time intervals farther apart than their duration are independent (independent increments).
\end{proof}

\begin{theorem}[Conditional certification with bursts and chip events, any linear decoding structure]\label{thm:st-cert}
Let the true noise be $\Nn_{\rm st}$ with $p_\ell=\varphi_\ell(p)$ (nondecreasing), flags $e_\ell=e$, Poisson burst activations $\rho_z=r w_z$, arbitrary burst strengths, and a chip layer of rate $r_c$ (Definition~\ref{def:events}). Suppose $\Prob[p\le\bar p,\ e\le\bar e,\ r\le\bar r,\ r_c\le\bar r_c]\ge1-\delta$ (for example from Lemmas~\ref{lem:flags}, \ref{lem:st-robust}, \ref{lem:robust}(c)), and that the region sizes and the coverage $\omega=\max_\ell\sum_{z\ni\ell}w_z$ obey $V_{\max}\le\bar V$, $\omega\le\bar\omega$. Then with probability $\ge1-\delta$
\[
\err\;\le\;\bar q_c+(1-\bar q_c)\min\Big\{1,\tfrac12\widetilde W^{\mathcal L}\big(\bar B\big)\Big\},
\]
where $\bar B$ is the $\bar B$ of Theorem~\ref{thm:st-burst} at $(\bar p,\bar e,\bar r\bar\omega,\bar V)$ (any $t\in(0,\tfrac12]$).\hfill\st{PROVED}
\end{theorem}
\begin{proof}
As for Theorem~\ref{thm:burst-cert}. Raising $p_\ell$ is superposing independent local depolarizing noise on unflagged locations (the composition rule above), raising $e$ is superposing flagged erasure, and raising $r,r_c$ is Poisson superposition; all are simulable reductions (Proposition~\ref{prop:rate-mono}, whose proof does not use code capacity). Then Lemma~\ref{lem:floor} and Theorem~\ref{thm:st-burst}; $\bar m$ and $\bar\kappa$ are evaluated at the true $p_\ell\le\varphi_\ell(\bar p)$ after the first step, and $\bar B$ is nondecreasing in $\bar\rho$ and $V_{\max}$.
\end{proof}

\noindent What is not done: $\widetilde W^{\mathcal L}$ of a circuit-level detector error model is not computed (it counts all undetectable logical fault sets by location weight; for $d\ge5$ this needs a transfer-matrix or path-counting bound, the circuit analogue of Theorem~\ref{thm:mwpm-peierls}); and $\bar B$ is below the radius of convergence of $\widetilde W^{\mathcal L}$ only for small $p$. Lemma~\ref{lem:st-robust} gives a valid but generally loose $\bar p$ (one detector, conservative under contamination).

\subsection{Tightening the code-capacity certificate (O7)}\label{sec:o7}

\begin{proposition}[A closed-form ML certificate for every distance]\label{prop:ml-peierls}
For the rotated surface code under $\Nn_{\rm cc}(p,e)$, every odd $d\ge3$, and $3B_M<1$,
\[
\err_d(p,e)\;\le\;\min\Big\{\tfrac12\widetilde W_d\big(B(p,e)\big),\ \frac{2dB_M(3B_M)^{d-1}}{1-3B_M}\Big\}.
\]
Both terms are nondecreasing in $p$ and $e$, so Theorem~\ref{thm:cc-cert} holds with this minimum evaluated at $(\bar p,\bar e)$. The second term needs no weight enumerator and is explicit for every $d$.\hfill\st{PROVED}
\end{proposition}
\begin{proof}
$\err_d$ is the failure probability of the optimal decoder, hence at most that of the MWPM decoder of Theorem~\ref{thm:mwpm-peierls}. Monotonicity: Corollaries~\ref{cor:product}, \ref{cor:mwpm-cert}.
\end{proof}

\begin{proposition}[Self-avoiding-walk refinement of the Peierls bound]\label{prop:saw}
Let $c_m$ be the number of self-avoiding walks of $m$ steps from the origin of $\mathbb Z^2$, $\mu_K=c_K^{1/K}$ and $\gamma_K=\max_{0\le r<K}c_r\mu_K^{-r}$. If $\mu_KB_M<1$, then for every odd $d\ge3$
\[
\pL^{\rm MWPM}(d;p,e)\;\le\;4d\,\gamma_K\,B_M^2\,\frac{(\mu_KB_M)^{d-2}}{1-\mu_KB_M},
\]
and the same bound holds for $\err_d$.\hfill\st{PROVED}
\end{proposition}
\begin{proof}
In the proof of Theorem~\ref{thm:mwpm-peierls} replace the count $d\,3^{\ell-1}$ of crossing paths of length $\ell$. The $Z$ checks have corners $(r,c)$ with $r+c$ odd, and two of them share a qubit iff their corners are diagonal neighbours, in which case they share exactly one; in the coordinates $(r+c,r-c)$ the check graph is a subgraph of a square lattice. A crossing path is a top qubit ($d$ choices), which fixes its check $v_1$; a self-avoiding walk of $\ell-2$ steps on the check graph from $v_1$ ($\le c_{\ell-2}$ choices; the qubit between consecutive checks is unique); and a bottom qubit of the last check ($\le2$ choices). So there are at most $2d\,c_{\ell-2}$ paths. Submultiplicativity $c_{m+n}\le c_mc_n$ \cite{MS} gives $c_m\le\gamma_K\mu_K^m$, and $\sum_{\ell\ge d}2d\,\gamma_K\mu_K^{\ell-2}B_M^\ell=2d\gamma_KB_M^2(\mu_KB_M)^{d-2}/(1-\mu_KB_M)$; the factor $2$ for the $Z$ part gives the claim.
\end{proof}

\noindent With the exact enumerations $c_{10}=44100$ and $c_{20}=897{,}697{,}164$ \cite{MS} one gets $\mu_{10}\approx2.914$, $\mu_{20}\approx2.803$, so at $e=0$ the condition $\mu_KB_M<1$ reads $p<4.56\%$ ($K=10$, $\gamma_{10}\approx1.46$) and $p<4.93\%$ ($K=20$), against $4.29\%$ for the factor $3$; no Peierls bound of this kind can go beyond $p<5.6\%$ (the connective constant $\mu\approx2.638$). Check H2 of \texttt{checks/handproof\_checks.py} enumerates $c_m$ for $m\le14$ (they agree with the cited values) and recomputes these ranges and $\gamma_{10}=1.455$; $c_{20}$ is cited, not enumerated. Check H3 finds the bounds above every MWPM and ML data point of P1 at which they are valid.
\emph{Status of O7.} Proposition~\ref{prop:ml-peierls} removes the practical limitation of Theorem~\ref{thm:cc-cert} (weight enumerators only for $d\le5$) and Proposition~\ref{prop:saw} moves the range of validity from $4.29\%$ to about $4.9\%$; the true ML threshold is $18.9\%$. Closing that gap needs a different argument (for example a rigorous renormalization bound, or a rigorous way to extrapolate exact small-$d$ failure rates); O7 stays open in that sense.

\subsection{Bursts by the Peierls route: no floor, and no strength floor needed (O3, O2)}\label{sec:peierls-bursts}

The union bound over logical elements cannot say anything as $d\to\infty$ (Proposition~\ref{prop:lower}(b)). The Peierls bound of \S\ref{sec:mwpm-cert} counts crossing paths only, and it carries over to bursts.

\begin{theorem}[Peierls bound with bursts]\label{thm:peierls-bursts}
Take the rotated surface code under the code-capacity burst noise of Definition~\ref{def:cc-bursts} (background $\Nn_{\rm cc}(p,e)$, possibly with qubit-dependent rates $p_j\le p$; regions of size $\le V$ and coverage density $\bar\rho$; arbitrary burst strengths), decoded by the MWPM decoder of Theorem~\ref{thm:mwpm-peierls}. For $\varsigma\ge1$ let $q=\tfrac23p$,
\[
m(s)=e+(1-e)\Big[\frac{1-s}{\varsigma}+s\,\varsigma\Big],\qquad\kappa=\frac{m(1/2)}{m(q)},\qquad B_{M,\rm b}(\varsigma)=m(q)\exp\Big(\bar\rho\,\frac{\kappa^V-1}V\Big).
\]
If $3B_{M,\rm b}<1$, then for every odd $d\ge3$
\[
\err_d\;\le\;\pL^{\rm MWPM}\;\le\;\frac{2d\,B_{M,\rm b}(3B_{M,\rm b})^{d-1}}{1-3B_{M,\rm b}},
\]
and the counts of Proposition~\ref{prop:saw} and Lemma~\ref{lem:row-count} may replace $d\,3^{\ell-1}$. At $\bar\rho=0$ and $\varsigma=\sqrt{(1-q)/q}$, $B_{M,\rm b}=B_M$. In particular bursts in regions of bounded size lower the exponent to at least $\ln\frac1{3B_{M,\rm b}}>0$ and produce no $d$-independent floor; this proves the O3 statement at code capacity.\hfill\st{PROVED}
\end{theorem}
\begin{proof}
The first step of the proof of Theorem~\ref{thm:mwpm-peierls} is deterministic: a failure of the $X$ part gives a crossing path $\gamma$, $|\gamma|\ge d$, on which the unflagged qubits with an $X$-type error are at least half of the unflagged qubits. For a fixed $\gamma$, $\mathbf 1[\mathcal B_\gamma]\le\prod_{j\in\gamma}\phi_j$ with $\phi_j=1$ on flagged qubits and $\phi_j=\varsigma^{2x_j-1}$ otherwise. Given the burst activations the qubits are independent; an unflagged qubit at depolarizing rate $s_j$ has $x_j\sim{\rm Bernoulli}(\tfrac23s_j)$ with $\tfrac23s_j\in[0,\tfrac12]$, so $\E[\phi_j\mid\text{activations}]=m(\tfrac23s_j)$, and $m$ is nondecreasing because $\varsigma\ge1/\varsigma$. Hence the factor is at most $m(q)$ on qubits not hit by an active region and at most $m(\tfrac12)$ on hit ones, and Steps~3--4 of the proof of Theorem~\ref{thm:burst-product} give $\Prob[\mathcal B_\gamma]\le B_{M,\rm b}^{|\gamma|}$. Sum over crossing paths as in Theorem~\ref{thm:mwpm-peierls}; ML is at most MWPM.
\end{proof}

\noindent The bound is crude: the optimal $\varsigma$ is much smaller than $\sqrt{(1-q)/q}$ once $\bar\rho>0$, because $\kappa^V$ grows like $\varsigma^V$. For example, at $p=0.01$, $e=0$, $V=4$ the condition $3B_{M,\rm b}<1$ needs $\bar\rho$ of order $10^{-5}$ or less. The statement is qualitative: an exponent that stays positive, no floor.

\begin{theorem}[Weak bursts need no rate bound: a certificate without a strength floor]\label{thm:weak-bursts}
Take the noise of Theorem~\ref{thm:peierls-bursts} with Poisson activations: region $z$ ($|z|\le V$) has events of rate $r_z$ per shot, each with a random strength $\theta\in[0,1]$ drawn independently from an unknown law $\nu_z$, and an event uniformizes each qubit of $z$ independently with probability $\theta$ (this is $\Dep_{3\theta/4}$ per qubit). No lower bound on $\theta$ is assumed. Fix any analysis threshold $\theta_0\in(0,1]$ and call events with $\theta\ge\theta_0$ strong.
\begin{enumerate}[(a)]
\item (Thinning.) The noise equals: an independent product background in which qubit $j$ has depolarizing rate $p_j$ with $\lambda(p_j)=\lambda(p)e^{-R_j}$, plus independent Poisson layers that uniformize sets $A$ with $|A|\ge2$, at rates $R_A$. Here $R_A=\sum_{z\supseteq A}r_z\E_{\nu_z}[\theta^{|A|}(1-\theta)^{|z|-|A|}]$.
\item (The correlated part is controlled by the strong rate and the background.) With $\bar\rho_m=\max_j\sum_{A\ni j,|A|\ge2}R_A$ and $\bar r_s=\max_j\sum_{z\ni j}r_z\nu_z([\theta_0,1])$,
\[
\bar\rho_m\le\bar r_s+(V-1)\,\theta_0(1-\theta_0)^{1-V}\max_j\ln\frac1{\lambda(p_j)}.
\]
\item (Data.) For each qubit $j$ pick a check $g(j)\ni j$. For any check $g$ containing $j$, on shots with no flag on $\supp g$: $\Prob[\text{$g$ fires}]=\tfrac12\big(1-\chi\prod_{i\in g}\lambda(p_i)\big)$ with $\chi\in(0,1]$, so an upper confidence limit $\bar P_g$ gives $\lambda(p_j)\ge1-2\bar P_g$. For a statistic $T$ equal to the parity of a check $g_W$, and regions $z$ with $|z\cap\supp g_W|\ge k$, the efficiency for strong events is at least $\eta(\theta_0)=\tfrac12\big(1-(1-\theta_0)^k\big)$, with no unknown parameter; so with an upper limit $\bar P_T$, simultaneously for all $\theta_0$, the strong rate of those regions is at most $-\ln\big(1-\min\{1,\bar P_T/\eta(\theta_0)\}\big)$.
\item (Certificate.) On the event that all these confidence limits hold (probability $\ge1-\delta$ by a union bound), for every $\theta_0\in(0,1]$ and every $d$,
\[
\err_d\;\le\;\bar q_c+(1-\bar q_c)\min\Big\{1,\ \frac{2d\,\bar B(3\bar B)^{d-1}}{1-3\bar B}\Big\},
\]
where $\bar B$ is $B_{M,\rm b}$ of Theorem~\ref{thm:peierls-bursts} evaluated at $\bar p=\tfrac34(1-\min_j\underline\lambda_j)$, $\bar e$, and $\bar\rho=\bar r_s(\theta_0)+(V-1)\theta_0(1-\theta_0)^{1-V}\ln(1/\min_j\underline\lambda_j)$, and $\bar q_c$ is the chip-layer term of Theorem~\ref{thm:burst-cert}. The infimum over $\theta_0$ and $\varsigma$ may be taken after the data are seen.
\end{enumerate}
\hfill\st{PROVED}
\end{theorem}
\begin{proof}
(a) By the marking theorem, the events of region $z$ whose uniformized set is $A$ form independent Poisson processes of rates $r_z\E_{\nu_z}[\theta^{|A|}(1-\theta)^{|z|-|A|}]$; merging over $z$ keeps independence. Uniformizing a qubit absorbs every other Pauli on it, so only whether $A$ was uniformized at least once matters. For $|A|=1$ these layers are independent across qubits; qubit $j$ is uniformized with probability $1-e^{-R_j}$, which multiplies its depolarizing eigenvalue by $e^{-R_j}$.
(b) A strong event contributes at most its rate to every $\sum_{A\ni j}R_A$. For weak events of region $z\ni j$, the probability that $j\in A$ and $|A|\ge2$ is $\E[\theta(1-(1-\theta)^{|z|-1})]\le(V-1)\theta_0\E[\theta]$, while the probability that $A=\{j\}$ is $\E[\theta(1-\theta)^{|z|-1}]\ge(1-\theta_0)^{V-1}\E[\theta]$. So the weak contribution to $\sum_{A\ni j,|A|\ge2}R_A$ is at most $(V-1)\theta_0(1-\theta_0)^{1-V}R_j$, and $R_j=\ln(\lambda(p)/\lambda(p_j))\le\ln(1/\lambda(p_j))$.
(c) The parity of $g$ is the sum of independent contributions: the product background gives $\prod_i\lambda(p_i)$, and each active multi-set meeting $\supp g$ (and each chip event) makes it uniform, which gives $\chi\in(0,1]$. Since every $\lambda(p_i)\le1$, $\lambda(p_j)\ge\chi\prod_i\lambda(p_i)=1-2\Prob[\text{fires}]$. Given at least one strong event in such a region, some qubit of $z\cap\supp g_W$ is uniformized with probability at least $1-(1-\theta_0)^k$, and then the parity is exactly uniform; other noise and flags only add firings. So $\Prob[T=1]\ge\eta(\theta_0)\big(1-e^{-r_s}\big)$ for every $\theta_0$ at once, which is Lemma~\ref{lem:robust}(c).
(d) Superposing depolarizing noise on unflagged qubits and superposing erasure are free (Theorem~\ref{thm:monotone}), so we may raise every $p_j$ to $\bar p$ and $e$ to $\bar e$. Then apply Theorem~\ref{thm:peierls-bursts} to the product background plus the multi-set layer of (a), whose sets have size $\le V$ and coverage $\le\bar\rho$ by (b) and (c), and the floor of Lemma~\ref{lem:floor}.
\end{proof}

\noindent This closes the main open item of O2. Proposition~\ref{prop:no-rate-limit} says that the total burst rate cannot be bounded from data without a strength floor, and Theorem~\ref{thm:weak-bursts} shows that it does not need to be: the strong part is bounded with a certified efficiency, and the weak part shows up in the background estimate, up to a correlated remainder of relative size $(V-1)\theta_0$. The trade-off in $\theta_0$ (a smaller $\theta_0$ shrinks the remainder but weakens $\eta$) is optimized a posteriori. Not covered: weak \emph{chip-scale} events (with $V=n$ the remainder term is useless unless $\theta_0\lesssim1/n$), and the circuit level, where (a)--(c) carry over to the locations of Definition~\ref{def:local-dep} (Lemma~\ref{lem:st-robust}) but no Peierls bound is available, so only the finite-$d$ bound of Theorem~\ref{thm:st-burst} can be used.

\subsection{Shape monotonicity in the region and duration directions (O1)}\label{sec:o1-more}

\begin{observation}[$\err$ can decrease when the burst region grows]\label{obs:O1-region}
On the $d=3$ rotated surface code with background $p=\tfrac1{200}$, $e=0$, one region activated with probability $\rho=\tfrac1{20}$ and strength $p_b=\tfrac34$: enlarging the region from the L-shaped corner $\{0,1,3\}$ to the plaquette $\{0,1,3,4\}$ lowers $\err$ from $5.754\times10^{-3}$ to $4.347\times10^{-3}$ ($-24.5\%$). Among all regions of the $3\times3$ lattice and the parameters $p\in\{0.005,0.02\}$, $\rho\in\{0.05,0.3\}$, $p_b\in\{0.3,0.75\}$, adding one qubit lowers $\err$ in $656$ cases ($84$ with both regions connected); along balls of growing radius around a qubit no decrease was found.\hfill\st{PROVED} (exact rational computation, check M7)
\end{observation}

\begin{observation}[$\err$ can decrease when the burst lasts longer]\label{obs:O1-duration}
For the bit-flip repetition code of distance $3$ over $3$ rounds with perfect measurements, background $p=\tfrac1{100}$ per data qubit and round, and one burst with probability $\rho=\tfrac1{20}$ starting at a uniformly random round on data qubits $\{0,1\}$ with flip probability $p_b=\tfrac12$: a duration of two rounds gives a smaller $\err$ than one round ($1.0591\times10^{-2}$ against $1.3613\times10^{-2}$, $-22.2\%$). The drop persists with measurement errors of probability $0.02$ in the first two rounds ($-22.7\%$, floating point); in $14$ of $128$ tested settings a longer burst gives a smaller $\err$.\hfill\st{PROVED} (exact rational computation, check M7)
\end{observation}

\noindent So O1 has a negative answer in all three shape directions. A plausible reading (not a proof): a stronger, larger or longer burst makes the hit qubits closer to uniform, and a uniform qubit carries no misleading evidence, whereas a partially noisy one biases the decision. Theorems~\ref{thm:burst-product}, \ref{thm:peierls-bursts} and~\ref{thm:weak-bursts} are uniform in the strength and monotone in $(V,\bar\rho)$, so they need no shape monotonicity.

\subsection{A Peierls bound for matching at circuit level (O5, O3, O2)}\label{sec:circuit-peierls}

The Peierls bounds of \S\ref{sec:mwpm-cert} and \S\ref{sec:peierls-bursts} use three facts: a decoder failure leaves a heavy crossing path; the errors on different edges are independent; and the crossing paths can be counted. In a circuit-level detector error model the second fact fails as stated, because one fault flips several detectors and stim splits it over several edges of the matching graph. The third fact needs the actual space-time graph.
This subsection restores the second fact exactly for the standard surface-code memory circuits, by re-decomposing the faults. It replaces the count by a sum over non-backtracking walks, which is computed on the graph. The bound is rigorous and computable for any detector error model that passes two checks (flag consistency and balance). For stim's rotated memory circuit it is non-vacuous at $p=10^{-3}$, about $40$--$240$ times the simulated MWPM failure rate at $d\le7$, and it gives a certified suppression factor for every $d$.
Fowler~\cite{Fowler} proved that matching has a positive threshold under circuit-level noise by a related path-counting argument. What is added here is a bound that is evaluated on a given detector error model and has explicit constants.

\begin{definition}[Observable graph and graph decompositions]\label{def:obs-graph}
A detector error model has independent mechanisms $m$ with probabilities $p_m\in[0,\tfrac12]$, detector sets $D_m$ and observable flips $L_m\in\F$ (one observable).
A \emph{decoding graph} $G$ has as vertices the detectors and one boundary vertex $b$. Every edge $e$ joins two detectors, or a detector and $b$, and carries a flag $o_e\in\F$ and a weight $w_e\ge0$. For an edge set $F$, $\partial F$ is the set of detectors of odd degree in $F$ and $o(F)=\sum_{e\in F}o_e$.
The \emph{observable graph} $G_{\rm o}$ consists of the connected components of $G-b$ that contain an edge with $o_e=1$, together with their edges to $b$. A \emph{graph decomposition} assigns to every mechanism an edge set $\Gamma(m)$ of $G_{\rm o}$ with $\partial\Gamma(m)=D_m\cap V(G_{\rm o})$ and $o(\Gamma(m))=L_m$.
Stim's \texttt{decompose\_errors=True} gives one, with $G$ the graph of its components, provided that no two components with the same detectors carry different flags. Any other decomposition may be used. In particular, a mechanism whose detectors in $G_{\rm o}$ are those of one edge $e$ of $G_{\rm o}$ with $o_e=L_m$ may be given $\Gamma(m)=\{e\}$; we call this \emph{re-decomposition}.
$G_{\rm o}$ is \emph{balanced} if every cycle of $G_{\rm o}-b$ has $o=0$. Then there is a function $\phi$ on the detectors of $G_{\rm o}$ with $o_{xy}=\phi(x)+\phi(y)$ on every edge (one $\phi$ per component), and the detectors $u$ with an edge to $b$ fall into two \emph{sides} $\sigma(u)=o_{ub}+\phi(u)\in\F$.
A mechanism is \emph{exclusive} if no simple cycle $\gamma$ with $o(\gamma)=1$ contains two edges of $\Gamma(m)$. Examples: $|\Gamma(m)|=1$, or all edges of $\Gamma(m)$ go to $b$ from one side.
\end{definition}

\begin{theorem}[Circuit-level Peierls bound for matching]\label{thm:circuit-peierls}
In the setting of Definition~\ref{def:obs-graph}, let $G_{\rm o}$ be balanced and let $\Gamma$ be a graph decomposition. The decoder outputs, for the syndrome $s$, an edge set $C$ of $G$ of minimum weight with $\partial C=s$ (ties broken arbitrarily) and predicts $o(C)$.
For an edge $e$ of $G_{\rm o}$ let $q_e=\tfrac12\big(1-\prod_{m}(1-2p_m)\big)$, the product over the exclusive $m$ with $e\in\Gamma(m)$. For the other mechanisms $m\ni e$ let $k_m=|\Gamma(m)|$. For $\lambda\ge0$ put $a_e=e^{2\lambda w_e}$ and
\[
\beta_e(\lambda)=e^{-\lambda w_e}\,(1-q_e+q_ea_e)\prod_{m\ \text{not exclusive},\ e\in\Gamma(m)}\big(1+p_m^{1/k_m}(a_e-1)\big).
\]
Let $\mathcal C$ be the set of simple cycles of $G_{\rm o}$ through $b$ whose two edges at $b$ end on different sides.
\begin{enumerate}[(a)]
\item For every $\lambda\ge0$, $\ \pL^{\rm MWPM}\le\sum_{\gamma\in\mathcal C}\prod_{e\in\gamma}\beta_e(\lambda)$.
\item Let $B$ be the non-backtracking matrix of $G_{\rm o}-b$ on directed edges, with entries $B_{(x\to y),(y\to z)}=\beta_{yz}$ for $z\ne x$ and $0$ otherwise. Put $\mathrm{start}_{(u\to y)}=\beta_{ub}\beta_{uy}$ if $\sigma(u)=0$ and $\mathrm{end}_{(x\to v)}=\beta_{vb}$ if $\sigma(v)=1$, both $0$ otherwise. If the spectral radius satisfies $\rho(B)<1$, the sum in (a) is at most $\mathrm{start}^{\mathsf T}(I-B)^{-1}\mathrm{end}$.
\item $\err\le\pL^{\rm MWPM}$. For fixed weights, both bounds are nondecreasing in every $p_m\in[0,\tfrac12]$, so they hold with each $p_m$ replaced by an upper limit $\bar p_m\le\tfrac12$.
\item (Bursts.) Let the mechanisms be the locations of Definition~\ref{def:local-dep} with $K=2$ and no flags, and let all of them be exclusive. Burst regions $z$ are sets of mechanisms with independent activations, $\Prob[N_z\ge1]\le\rho_z$. An activation raises the probability of each mechanism of $z$ to an arbitrary value in $[p_m,\tfrac12]$, independently across mechanisms given the activations.
Let $E(z)$ be the edges $e$ with $e\in\Gamma(m)$ for some $m\in z$, and put $V_E=\max_z|E(z)|$, $\bar\rho_E=\max_e\sum_{z:\,E(z)\ni e}\rho_z$, $\kappa_e=\tfrac12e^{-\lambda w_e}(1+a_e)/\beta_e(\lambda)\ge1$ and $\bar\kappa=\max_e\kappa_e$.
Then (a) and (b) hold, uniformly in the burst strengths, with every $\beta_e$ replaced by $\beta_e\,\Xi$, where $\Xi=\exp\big(\bar\rho_E(\bar\kappa^{V_E}-1)/V_E\big)$.
\item (Rounded weights.) Suppose the decoder minimizes other weights $\tilde w_e$ instead of $w_e$, and $|\tilde w_e/s-w_e|\le\delta$ on every edge of $G_{\rm o}$ for some $s>0$. Then (a)--(d) hold with every $\beta_e(\lambda)$ replaced by $e^{\lambda\delta}\beta_e(\lambda)$ ($\kappa_e$ unchanged).
Pymatching~2 \cite{SB} minimizes $\tilde w_e=2\,\mathrm{round}(\varkappa w_e)$, where $w_e$ are its floating-point weights and $\varkappa=(2^{24}-1)/\max_e|w_e|$, the maximum over all edges ($\varkappa=1$ if every $w_e$ is an integer). So $s=2\varkappa$ and $\delta=\max_e|w_e|/(2(2^{24}-1))$, or $\delta=0$.
\end{enumerate}
\hfill\st{PROVED}
\end{theorem}
\begin{proof}
\emph{Failure leaves a heavy odd cycle.}
Let $f_m$ indicate that $m$ fires, and let $X=\bigoplus_{f_m=1}\Gamma(m)$ (symmetric difference). Then $\partial X=s\cap V(G_{\rm o})$, and $o(X)=\sum_mf_mL_m$ is the actual flip of the observable.
The boundary vertex imposes no constraint, so the minimization splits over the components of $G-b$. Hence $C_{\rm o}=C\cap E(G_{\rm o})$ is a minimum-weight edge set of $G_{\rm o}$ with $\partial C_{\rm o}=s\cap V(G_{\rm o})$, and the other edges of $C$ have $o=0$. So the decoder fails iff $o(X\oplus C_{\rm o})=1$.
$Y=X\oplus C_{\rm o}$ has even degree at every detector, hence also at $b$, so it is an edge-disjoint union of simple cycles; on failure one of them, $\gamma$, has $o(\gamma)=1$.
$C_{\rm o}\oplus\gamma$ has the same boundary as $C_{\rm o}$, so minimality gives $w(\gamma\cap C_{\rm o})\le w(\gamma\setminus C_{\rm o})$. Every edge of $\gamma\subseteq Y$ lies in exactly one of $X$ and $C_{\rm o}$, so $\gamma\setminus C_{\rm o}=\gamma\cap X$ and $w(\gamma\cap X)\ge\tfrac12w(\gamma)$.
By balance $\gamma$ passes through $b$. If its edges at $b$ are $ub$ and $vb$, then $o(\gamma)=o_{ub}+\phi(u)+\phi(v)+o_{vb}=\sigma(u)+\sigma(v)$. So $\gamma\in\mathcal C$.

\emph{One cycle.}
Fix $\gamma\in\mathcal C$ and let $x_e=\mathbf 1[e\in X]$. Then $\mathbf 1[w(\gamma\cap X)\ge\tfrac12w(\gamma)]\le\prod_{e\in\gamma}e^{-\lambda w_e}a_e^{x_e}$.
Let $y_e$ be the parity of the firing exclusive mechanisms with $e\in\Gamma(m)$. Then $x_e\le y_e+\sum f_m$, the sum over the non-exclusive $m$ with $e\in\Gamma(m)$, and $a_e\ge1$.
An exclusive mechanism has at most one edge on $\gamma$, so the $y_e$, $e\in\gamma$, are functions of disjoint sets of mechanisms. They are therefore independent of each other and of the non-exclusive mechanisms, with $\Prob[y_e=1]=q_e$. Hence
\[
\Prob\big[w(\gamma\cap X)\ge\tfrac12w(\gamma)\big]\le\prod_{e\in\gamma}e^{-\lambda w_e}(1-q_e+q_ea_e)\prod_{m\ \text{not excl.}}\Big(1-p_m+p_m\prod_{e\in\gamma\cap\Gamma(m)}a_e\Big).
\]
\emph{Splitting.} If $t_i\ge0$, $r_i\in[0,1]$ and $\prod_ir_i\ge p$, then $1+p\big(\prod_i(1+t_i)-1\big)\le\prod_i(1+r_it_i)$. Expanding both sides, the coefficient of $\prod_{i\in S}t_i$ is $p$ on the left and $\prod_{i\in S}r_i\ge\prod_ir_i\ge p$ on the right.
With $r_e=p_m^{1/k_m}$ for $e\in\gamma\cap\Gamma(m)$ we have $\prod r_e=p_m^{|\gamma\cap\Gamma(m)|/k_m}\ge p_m$. So the last product is at most $\prod_{e\in\gamma}\prod_{m\ni e}(1+p_m^{1/k_m}(a_e-1))$, and the probability is at most $\prod_{e\in\gamma}\beta_e(\lambda)$. A union bound over $\mathcal C$ gives (a).

\emph{Walks.}
Orient $\gamma\in\mathcal C$ from its end $u$ on side $0$. Then $\gamma$ is $b,u=x_0,x_1,\dots,x_n=v,b$ with $n\ge1$, and the path $x_0\dots x_n$ is simple, hence a non-backtracking walk in $G_{\rm o}-b$. Its weight $\prod_{e\in\gamma}\beta_e$ is $\mathrm{start}_{(x_0\to x_1)}\prod_{i=1}^{n-1}B_{(x_{i-1}\to x_i),(x_i\to x_{i+1})}\,\mathrm{end}_{(x_{n-1}\to x_n)}$, and different cycles give different walks. All terms are nonnegative, so the sum in (a) is at most $\sum_{n\ge0}\mathrm{start}^{\mathsf T}B^n\mathrm{end}=\mathrm{start}^{\mathsf T}(I-B)^{-1}\mathrm{end}$ when $\rho(B)<1$. This proves (b).

(c) $\err$ is the Bayes error (Lemma~\ref{lem:decoding}), so no decoder beats ML. $1-q+qa_e$ is nondecreasing in $q$, and $q_e$ is nondecreasing in each $p_m\le\tfrac12$. The factors $1+p^{1/k}(a_e-1)$ are nondecreasing in $p$. The Neumann series, and $\rho(B)$, are nondecreasing in every entry. The weights are the decoder's and do not change.

(d) Given the activations the mechanisms are independent. Edges not hit by an active region keep $q_e$; hit edges have a parity probability in $[q_e,\tfrac12]$, and $\tfrac12$ gives the factor $\tfrac12e^{-\lambda w_e}(1+a_e)=\kappa_e\beta_e$. So the probability for $\gamma$ given the activations is at most $\prod_{e\in\gamma}\beta_e\prod_z\prod_{e\in\gamma\cap E(z)}\kappa_e^{\mathbf 1[N_z\ge1]}$.
Average over the independent activations, and use $1+x\le e^x$, $\bar\kappa^k-1\le k(\bar\kappa^{V_E}-1)/V_E$ for $0\le k\le V_E$, and $\sum_z\rho_z|\gamma\cap E(z)|\le|\gamma|\bar\rho_E$. This gives the factor $\Xi^{|\gamma|}$, as in Steps~3--4 of Theorem~\ref{thm:burst-product}.

(e) Only the minimality step changes. Write $\tilde w_e=s(w_e+\epsilon_e)$ with $|\epsilon_e|\le\delta$. Minimality for $\tilde w$ gives $\tilde w(\gamma\cap C_{\rm o})\le\tilde w(\gamma\setminus C_{\rm o})$, hence $w(\gamma\cap C_{\rm o})-w(\gamma\setminus C_{\rm o})\le\sum_{e\in\gamma}|\epsilon_e|\le\delta|\gamma|$ and $w(\gamma\cap X)\ge\tfrac12w(\gamma)-\tfrac12\delta|\gamma|$. Then $\mathbf 1[\,\cdot\,]\le\prod_{e\in\gamma}e^{\lambda(\delta-w_e)}a_e^{x_e}$, and the rest of the proof is unchanged with the extra factor $e^{\lambda\delta}$ per edge. The value of $\tilde w$ for pymatching is read off its source (\texttt{sparse\_blossom/driver/user\_graph.h}, \texttt{user\_graph.cc}, version 2.4.0): weights are scaled by $\varkappa$, rounded, and doubled; rounding moves $\varkappa w_e$ by at most $\tfrac12$.
\end{proof}

\noindent At code capacity, $G_{\rm o}$ is the matching graph of Theorem~\ref{thm:mwpm-peierls}, every mechanism has one edge, and $w_e=\ln\frac{1-q}q$, $\lambda=\tfrac12$ give $\beta_e=2\sqrt{q(1-q)}$. Part (b) then replaces the count $d\,3^{\ell-1}$ by the exact non-backtracking walk sum.
The bound is log-convex in $\lambda$: each $\beta_e$ is a positive combination of exponentials in $\lambda$, and sums and products preserve log-convexity. So the minimization over $\lambda$ is one-dimensional and convex; its optimum is near $\tfrac12$ when the weights are log-likelihood ratios.
Pymatching~\cite{SB} computes an exact minimum-weight correction for its weights after converting them to integers; part (e) accounts for the rounding. For the circuits below $\delta\approx2\times10^{-7}$, so the factor $e^{\lambda\delta|\gamma|}$ changes no printed digit, but all numbers include it. Check N2 tests the premise of (e) on sampled faults: every cycle of $X\oplus C$ satisfies $w(\gamma\cap C)\le w(\gamma\setminus C)+\delta|\gamma|$ (the largest excess found is $10^{-16}$, i.e. floating-point error).
The computation uses $(I-B)z=\mathbf 1$: if $z>0$, then $\rho(B)\le1-1/\max z<1$ (Collatz--Wielandt), so the same factorization both certifies convergence and evaluates the bound.

\begin{proposition}[A bound for every $d$, and a certified exponent]\label{prop:circuit-lattice}
In the circuit-level setting of Proposition~\ref{prop:circuit-numbers}, place the detector of the $Z$ plaquette with centre $(x,y)$ at round $t$ at the point $(x,y,t)$ (stim's coordinates). Fix $\lambda=\tfrac12$ and assume:
\begin{enumerate}[(H1)]
\item every edge of $G_{\rm o}-b$ joins two detectors whose difference is one of the twelve offsets $\pm(2,2,0)$, $\pm(2,-2,0)$, $\pm(0,0,1)$, $\pm(2,2,-1)$, $\pm(2,-2,-1)$, $\pm(4,0,-1)$, and its factor $\beta_e$ (including the rounding factor $e^{\lambda\delta}$ of Theorem~\ref{thm:circuit-peierls}(e)) is at most an envelope value $\hat\beta(v)$ of its offset $v$, which does not depend on $d$; every boundary factor is at most $\hat\beta_b$;
\item the two sides are the rows $y=2$ and $y=2d-2$, each with $(d+1)^2/2$ detectors.
\end{enumerate}
For $\theta\ge0$ let $\hat B(\theta)$ be the non-backtracking matrix of the offsets, $\hat B(\theta)_{v,v'}=\hat\beta(v')e^{-\theta\Delta y(v')/2}$ for $v'\ne-v$. Let $\hat\rho(\theta)$ be its spectral radius, $R(\theta)$ the ratio of the largest to the smallest entry of its right Perron vector, and $\hat s(\theta)=\sum_v\hat\beta(v)e^{-\theta\Delta y(v)/2}$. If $\hat\rho(\theta)<1$, then for every odd $d\ge5$
\[
\err_d\;\le\;\pL^{\rm MWPM}(d)\;\le\;\frac{(d+1)^2}{2}\,\hat\beta_b^2\,\frac{\hat s(\theta)R(\theta)}{1-\hat\rho(\theta)}\,e^{-\theta(d-2)}.
\]
So $\liminf_d-\tfrac1d\ln\pL^{\rm MWPM}(d)\ge\theta^\ast=\sup\{\theta:\hat\rho(\theta)<1\}$, and the suppression factor per distance step of two is at least $\Lambda^\ast=e^{2\theta^\ast}$.
Numerically, (H1) and (H2) hold for $5\le d\le15$ with the same envelope and boundary factor (checks N5 and the experiment script). The values are:
$\theta^\ast=2.23,\ 1.83,\ 1.24,\ 0.671$ ($\Lambda^\ast=86,\ 39,\ 12.0,\ 3.83$) at $p=10^{-4},\ 2\times10^{-4},\ 5\times10^{-4},\ 10^{-3}$, and $\hat\rho(0)<1$ for $p<1.405\times10^{-3}$. The growth rate $\hat\rho(0)$ of the bulk alone reaches $1$ at $p=1.55\times10^{-3}$.
The same statement and proof hold for phenomenological noise, where the offsets are $\pm(2,\pm2,0)$ and $\pm(0,0,1)$. There $\theta^\ast=1.93,\ 1.53,\ 0.915$ at $p=10^{-3},\ 2\times10^{-3},\ 5\times10^{-3}$, and $\hat\rho(0)<1$ for $p<1.01\times10^{-2}$ ($1.32\times10^{-2}$ for the bulk alone).
\hfill\st{PROVED} for $5\le d\le15$ (computer-assisted); for $d>15$ conditional on (H1)--(H2), which rest on the translation structure of the generated circuit and are not checked there
\end{proposition}
\begin{proof}
Orient every $\gamma\in\mathcal C$ from its end $u$ on the row $y=2d-2$. Its path $u\to v$ in $G_{\rm o}-b$ is a non-backtracking walk. By (H1) every step changes $y$ by $0$ or $\pm2$, and by (H2) the steps add up to $\Delta y=-(2d-4)$. Hence
$\prod_{e}\beta_e\le\prod_e\hat\beta(v_e)=e^{-\theta(d-2)}\prod_e\hat\beta(v_e)e^{-\theta\Delta y(v_e)/2}$.
For fixed $u$, a walk is determined by its sequence of offsets, and non-backtracking walks give sequences with $v_{i+1}\ne-v_i$. So the sum over the walks from $u$ is at most
$e^{-\theta(d-2)}\sum_{v_1}\hat\beta_\theta(v_1)\sum_{n\ge0}(\hat B(\theta)^n\mathbf 1)_{v_1}$.
If $r>0$ is the right Perron vector, then $\hat B^n\mathbf 1\le\hat B^nr/\min r=\hat\rho^nr/\min r\le\hat\rho^nR\,\mathbf 1$, so this is at most $e^{-\theta(d-2)}\hat s(\theta)R(\theta)/(1-\hat\rho(\theta))$.
Multiply by the two boundary factors ($\le\hat\beta_b^2$), sum over the $(d+1)^2/2$ choices of $u$, and apply Theorem~\ref{thm:circuit-peierls}(a),(c).
\end{proof}

\begin{corollary}[Bursts at circuit level: no floor, and a certificate without a strength floor]\label{cor:circuit-bursts}
In the setting of Proposition~\ref{prop:circuit-lattice}, let bursts act on regions of mechanisms as in Theorem~\ref{thm:circuit-peierls}(d), with $V_E$ and $\bar\rho_E$ bounded. If $\Xi\,\hat\rho(\theta)<1$, the bound of Proposition~\ref{prop:circuit-lattice} holds with $\hat\beta$, $\hat\beta_b$ replaced by $\Xi\hat\beta$, $\Xi\hat\beta_b$ (and $\hat s$ by $\Xi\hat s$). In particular the failure probability still decays exponentially in $d$, and there is no $d$-independent floor; this is the O3 statement at circuit level.
Theorem~\ref{thm:weak-bursts} carries over with the same proof:
\begin{itemize}
\item qubits are replaced by mechanisms ($K=2$, $\lambda(p)=1-2p$), and Lemma~\ref{lem:st-robust} replaces the check-based limits in (c);
\item in (d), Theorem~\ref{thm:circuit-peierls}(c),(d) replaces Theorem~\ref{thm:peierls-bursts}, with $\bar\rho_E\le\max_e|\{m:e\in\Gamma(m)\}|\cdot\bar\rho$ and $V_E\le V$.
\end{itemize}
This gives a circuit-level certificate without a floor on the burst strength (O2).\hfill\st{PROVED} under (H1)--(H2)
\end{corollary}
\begin{proof}
The first part combines Theorem~\ref{thm:circuit-peierls}(d) with the proof of Proposition~\ref{prop:circuit-lattice}.
For the second part, check the steps of Theorem~\ref{thm:weak-bursts} at the level of mechanisms:
\begin{itemize}
\item (a) The thinning uses only the marking theorem and the fact that uniformizing a location absorbs every other fault on it; both hold for mechanisms ($K=2$).
\item (b) The coverage estimate is the same algebra with $\lambda(p)=1-2p$.
\item (c) The parity of a detector becomes uniform as soon as one mechanism in its support is uniformized.
\item (d) The background enters only through the monotone bound of Theorem~\ref{thm:circuit-peierls}(c). A multi-set layer $A$ hits the edges $E(A)$, with $|E(A)|\le|A|$, and an edge collects the layers of all its mechanisms.
\end{itemize}
\end{proof}

\noindent\emph{How tight.} At $p=10^{-3}$ the bound is $39$, $91$ and $240$ times the simulated MWPM failure rate at $d=3$, $5$ and $7$. Its certified suppression factor $\Lambda^\ast=3.8$ compares with a simulated $\Lambda\approx6$--$7$.
The looseness comes from the Peierls step (the sum over all subsets of a cycle, with a Chernoff factor for the condition ``half of the cycle in error''), not from the counting of cycles; Theorem~\ref{thm:geodesic} below removes part of it. At $d=3$ with two rounds, the non-backtracking walk sum exceeds the exact sum over the $29{,}138$ odd simple cycles by only $3\%$ (check N3). Walks that also avoid the last two or three vertices lower the bulk growth rate by only $3$--$5\%$.
\emph{Status.} This answers O5 at circuit level and for phenomenological noise (computable rigorous bounds for MWPM, hence for ML; the constants are still crude). It also gives the circuit-level forms of O3 and O2 under (H1)--(H2). What is not covered: erasure flags at circuit level, where the decoder changes its weights shot by shot; ML-specific (non-MWPM) bounds; and circuits whose decomposition is not balanced or leaves non-exclusive mechanisms (part (a) still applies, with the $p^{1/k}$ split, but no case was computed).

\paragraph*{Geodesic refinement.} The Peierls step of Theorem~\ref{thm:circuit-peierls} sums over all subsets $S=\gamma\cap X$ of an odd cycle. Minimality of the correction restricts the complement $\gamma\setminus S$ much more than the weight condition alone: each of its runs is a shortest path.

\begin{theorem}[Geodesic refinement of the circuit-level Peierls bound]\label{thm:geodesic}
In the setting of Theorem~\ref{thm:circuit-peierls}, let every mechanism be exclusive and let the decoder minimize weights $\tilde w_e>0$ (for pymatching, the integers $\tilde w_e=2\,\mathrm{round}(\varkappa w_e)$ of part (e)); put $\hat w=\tilde w/s$ for any $s>0$. A \emph{run} of an edge set $F\subseteq\gamma$ is a maximal sequence of consecutive edges of $\gamma$ in $F$; it is a path of $G_{\rm o}$ (it may pass through $b$).
\begin{enumerate}[(a)]
\item On failure there is $\gamma\in\mathcal C$ with $\gamma\subseteq X\oplus C_{\rm o}$, $S=\gamma\cap X\ne\emptyset$, $\tilde w(\gamma\setminus S)\le\tilde w(S)$, and every run of $\gamma\setminus S$ is a $\tilde w$-geodesic of $G_{\rm o}$ between its end points.
\item For $\lambda\ge0$ put $x_e=q_e e^{\lambda\hat w_e}$ and $c_e=(1-q_e)e^{-\lambda\hat w_e}$ (so $x_e+c_e=\beta_e(\lambda)$ when $\hat w=w$). Then
\[
\pL^{\rm MWPM}\le\sum_{\gamma\in\mathcal C}\ \sum_{S}\ \prod_{e\in S}x_e\prod_{e\in\gamma\setminus S}c_e,
\]
the inner sum over the nonempty $S\subseteq\gamma$ whose complement has only geodesic runs.
\item (Computable form.) On the vertices of $G_{\rm o}-b$ let $K_X(y,z)$ be the sum of $\prod x_e$ over non-backtracking walks of length $\ge1$ from $y$ to $z$ in $G_{\rm o}-b$, and $K_C(y,z)$, $y\ne z$, the sum of $\prod c_e$ over paths from $y$ to $z$ in $G_{\rm o}-b$ that are $\tilde w$-geodesics of $G_{\rm o}$. Let $\sigma_i(u)=x_{ub}$ if $u$ has an edge to $b$ and $\sigma(u)=i$, else $0$; $s_X=\sigma_0^{\mathsf T}(I+K_X)$, $e_X=(I+K_X)\sigma_1$; and let $s_C(z)$ (resp.\ $e_C(z)$) be the sum of $\prod c_e$ over the $\tilde w$-geodesics of $G_{\rm o}$ from $b$ to $z$ whose first edge ends on side $0$ (resp.\ $1$) and which avoid $b$ afterwards. With $M=\begin{pmatrix}0&K_C\\K_X&0\end{pmatrix}$, if $\rho(M)<1$ and the walk sums converge,
\[
\pL^{\rm MWPM}\le s_X\sigma_1+(s_X,\,s_C)\,(I-M)^{-1}\begin{pmatrix}e_C\\ e_X\end{pmatrix}.
\]
\item $\err\le\pL^{\rm MWPM}$, so (b) and (c) bound $\err$ as well.
\end{enumerate}
\hfill\st{PROVED} (draft; checks G1--G3)
\end{theorem}
\begin{proof}
(a) Take $\gamma$ as in the proof of Theorem~\ref{thm:circuit-peierls}, with the decoder's weights $\tilde w$ in the minimality step; this gives $\tilde w(\gamma\cap C_{\rm o})\le\tilde w(\gamma\cap X)$ without rounding slack, and $S\neq\emptyset$ because the weights are positive. Let $\sigma$ be a run of $\gamma\setminus S=\gamma\cap C_{\rm o}$ from $u$ to $v$, and $\sigma'$ any $u$--$v$ path of $G_{\rm o}$. The edge set $\sigma\oplus\sigma'$ has even degree at every vertex, so $C'=C_{\rm o}\oplus\sigma\oplus\sigma'$ has the syndrome of $C_{\rm o}$ ($b$ imposes no constraint). As $\sigma\subseteq C_{\rm o}$, $\tilde w(C')\le\tilde w(C_{\rm o})-\tilde w(\sigma)+\tilde w(\sigma')$, and minimality gives $\tilde w(\sigma)\le\tilde w(\sigma')$.

(b) Fix $\gamma$ and $S$. The event that $\gamma\cap X=S$ is the event $S\subseteq X$, $(\gamma\setminus S)\cap X=\emptyset$. As in Theorem~\ref{thm:circuit-peierls}, exclusivity makes the indicators $\mathbf 1[e\in X]$, $e\in\gamma$, independent with probabilities $q_e$, so this event has probability $\prod_{S}q_e\prod_{\gamma\setminus S}(1-q_e)$. On it, failure through $\gamma$ needs $\tilde w(\gamma\setminus S)\le\tilde w(S)$, whose indicator is at most $\exp\big(\lambda(\hat w(S)-\hat w(\gamma\setminus S))\big)$. By (a), the union over $\gamma$ and the admissible $S$ covers the failure event.

(c) Orient $\gamma\in\mathcal C$ from its end on side $0$, so it reads $b,u,\dots,v,b$, and cut it at $b$. Its edges fall into maximal runs of $S$ (X-runs) and of $\gamma\setminus S$ (C-runs), alternating; a run containing both edges at $b$ is cut into a first and a last piece. If $\gamma\subseteq S$, the term is that of an X-walk from $b$ ending with the edge $vb$: the sum of these terms is at most $s_X\sigma_1$. Otherwise the sequence is: a first run from $b$ (an X-run, counted in $s_X$, or a C-run, counted in $s_C$); interior runs avoiding $b$ that alternate in type (an X-run is a non-backtracking walk in $G_{\rm o}-b$, counted in $K_X$; a C-run is a sub-path of a geodesic, hence a geodesic, counted in $K_C$); and a last run into $b$ of the type opposite to the preceding one (counted in $e_C$ or $e_X$, by symmetry of geodesics and of walks). Distinct pairs $(\gamma,S)$ give distinct sequences. All terms are nonnegative, so the sum of (b) is at most the sum over all alternating sequences, which is the Neumann series of $M$; it converges to the stated expression when $\rho(M)<1$, certified as in Theorem~\ref{thm:circuit-peierls} by a positive solution of $(I-M)z=\mathbf 1$.
(d) As in Theorem~\ref{thm:circuit-peierls}(c).
\end{proof}

\paragraph*{Certificates from the data's own statistics.} The bounds above take the edge probabilities $q_e$ as given. On hardware they are unknown, and the published models are fits. The $p_{ij}$ method of~\cite{Google21} estimates them from the two-point correlations of the detection events. Under the assumption from which that estimator is derived, the estimate becomes a set of simultaneous confidence limits, and Theorems~\ref{thm:circuit-peierls} and~\ref{thm:geodesic} turn the limits into an upper bound on $\pL$ that holds with a stated probability over the data. The decoding model enters only through its graph, flags and weights, and the weights may themselves be fitted to the data.
\emph{Assumption (G).} Let $G_{\rm o}$ be the observable graph of a fixed decoding model (Definition~\ref{def:obs-graph}), balanced and with all mechanisms exclusive. Parallel components are merged into one edge, so $G_{\rm o}$ is simple: at most one edge joins two detectors, and each detector has at most one edge to $b$. On the detectors of $G_{\rm o}$, every shot is $X=\bigoplus_ef_e\,e$ with independent $f_e\sim\mathrm{Bernoulli}(q_e)$, $q_e\in[0,\tfrac12]$, and the observable flips by $o(\{e:f_e=1\})$. The shots are i.i.d.
For an assumption (T) of \emph{time stationarity}, call two edges of the same type if their end points have the same coordinate pattern up to a common time shift and lie in the same layer (first round, bulk, final data measurement). (T) states that edges of the same type have the same $q_e$.

\begin{proposition}[Data certificate]\label{prop:data-cert}
Assume (G). For a detector $i$ let $x_i$ be its detection event, $Z_i=(-1)^{x_i}$, $a_i=\E x_i$, and for an edge $e=ij$ let $c_{ij}=\E x_ix_j$, $s_e=1-2q_e$.
\begin{enumerate}[(a)]
\item ($p_{ij}$ identity.) For an edge $ij$ between two detectors, $\E Z_i\,\E Z_j=s_{ij}^2\,\E Z_iZ_j$, that is
\[
1-(1-2q_{ij})^2=\frac{4T_{ij}}{R_{ij}},\qquad T_{ij}=c_{ij}-a_ia_j=\mathrm{Cov}(x_i,x_j),\quad R_{ij}=1-2a_i-2a_j+4c_{ij}=1-2\Prob[x_i\ne x_j].
\]
For the edge $b_i$ from $i$ to $b$: $1-2a_i=s_{b_i}\prod_{e\ni i,\,e\ne b_i}s_e$.
\item (Propagation.) Suppose $T_{ij}\in[T^-,T^+]$ and $R_{ij}\in[R^-,R^+]$. Then $q_{ij}\le q^+:=\tfrac12\big(1-\sqrt{1-4T^+/R^-}\big)$ if $T^+>0$ and $4T^+<R^-$ ($q^+=0$ if $T^+\le0$, and $q^+=\tfrac12$ otherwise). Likewise $q_{ij}\ge q^-:=\tfrac12\big(1-\sqrt{1-4T^-/R^+}\big)$ if $T^->0$ and $q^-=0$ otherwise. If $a_i\in[a^-,a^+]$ and $q_e\in[q_e^-,q_e^+]$ for the other edges at $i$, then
\[
1-2q_{b_i}\ \ge\ \frac{1-2a^+}{\prod_{e}(1-2q_e^-)},\qquad 1-2q_{b_i}\ \le\ \frac{1-2a^-}{\prod_{e}(1-2q_e^+)},
\]
with the products over the edges $e\ni i$, $e\ne b_i$. These give $q_{b_i}^+$ and $q_{b_i}^-$, clipped to $[0,\tfrac12]$.
The same holds under (T) for a group $g$ of edges of one type, with $T,R$ replaced by their averages over $g$. At a boundary, $a_i$ is replaced by its average over $g$ and $\prod_e$ by the average over $i\in g$ of the products.
\item (Certificate.) Let the intervals for the $T$'s, $R$'s and boundary $a_i$'s hold simultaneously with probability $\ge1-\delta$. For example, take $M$ such statistics, each with a confidence interval at level $1-\delta/M$ (Bonferroni). Then with probability $\ge1-\delta$ over the data, simultaneously for all weights,
\[
\err\ \le\ \pL^{\rm MWPM}\ \le\ \min\big\{B_{\rm P}(q^+),\ B_{\rm G}(q^-,q^+)\big\}.
\]
Here $B_{\rm P}(q^+)$ is Theorem~\ref{thm:circuit-peierls}(b) evaluated at $q^+$, and $B_{\rm G}(q^-,q^+)$ is Theorem~\ref{thm:geodesic}(c) with $x_e=q_e^+e^{\lambda\hat w_e}$ and $c_e=(1-q_e^-)e^{-\lambda\hat w_e}$; both use the decoder's weights.
\end{enumerate}
\hfill\st{PROVED} under (G) and i.i.d.\ shots (and (T) for the pooled form)
\end{proposition}
\begin{proof}
(a) Write $Y_e=(-1)^{f_e}$, so $\E Y_e=s_e$ and $Z_i=\prod_{e\ni i}Y_e$; hence $\E Z_i=\prod_{e\ni i}s_e$, which is the boundary identity. Since $G_{\rm o}$ is simple, $ij$ is the only edge containing both $i$ and $j$. So $Z_iZ_j=\prod_{e\ni i,\,e\ne ij}Y_e\prod_{e\ni j,\,e\ne ij}Y_e$ is a product over distinct edges ($Y_{ij}^2=1$), and $\E Z_iZ_j=\E Z_i\E Z_j/s_{ij}^2$. Finally $\E Z_i=1-2a_i$, $\E Z_iZ_j=1-2a_i-2a_j+4c_{ij}$, and $(1-2a_i)(1-2a_j)=R_{ij}-4T_{ij}$.
(b) Under (G), $R_{ij}=\prod s_e$ over the edges at $i$ or $j$ other than $ij$, so $R_{ij}>0$ when all $q_e<\tfrac12$. Also $T_{ij}=q_{ij}(1-q_{ij})R_{ij}\ge0$. The map $(T,R)\mapsto\tfrac12(1-\sqrt{1-4T/R})$ is nondecreasing in $T$, and for $T\ge0$ nonincreasing in $R>0$. This gives $q^\pm$; the clipping only enlarges the interval. For the boundary, $s_{b_i}=(1-2a_i)/P_i$ with $P_i=\prod_e s_e\in\big[\prod(1-2q_e^+),\prod(1-2q_e^-)\big]$ (factors in $[0,1]$). The ratio is nonincreasing in $a_i$, and nonincreasing in $P_i$ while $1-2a_i>0$. Under (T), $\E\bar T_g=(1-s^2)\bar R_g/4$ and $1-2\bar a_g=s_{b}\,\overline{P}_g$ for the common $s$, so the same monotonicity applies to the averages.
(c) On the event that all intervals cover, (b) gives $q_e^-\le q_e\le q_e^+$ on every edge. Theorem~\ref{thm:circuit-peierls}(c) is nondecreasing in every $q_e$. Theorem~\ref{thm:geodesic} is not monotone in $q_e$ ($x_e$ increases and $c_e$ decreases), but its proof uses $q$ only through $\Prob[S\subseteq X,\ (\gamma\setminus S)\cap X=\emptyset]=\prod_{S}q_e\prod_{\gamma\setminus S}(1-q_e)\le\prod_Sq_e^+\prod_{\gamma\setminus S}(1-q_e^-)$. Every later step is a sum of nonnegative terms, nondecreasing in each $x_e$ and $c_e$. The event does not involve the weights.
\end{proof}

\noindent\emph{Statistics} (\texttt{src/lcd/analysis/data\_certificate.py}). The literal recipe (``cp'') takes Clopper--Pearson limits for every $a_i$ and $c_{ij}$ and propagates them by interval arithmetic: $T^+=c^+-a_i^-a_j^-$, $R^-=1-2a_i^+-2a_j^++4c^-$, and symmetrically. The default (``paired'') estimates the covariance directly. Pair the shots $(2k-1,2k)$; then $D=(x_i-x_i')(x_j-x_j')\in\{-1,0,1\}$ is i.i.d.\ with $\E D=2T_{ij}$. Its mean gets a betting confidence interval (the hedged-capital construction of Waudby-Smith and Ramdas, valid at every sample size by Ville's inequality), and $\Prob[x_i\ne x_j]$ and the boundary $a_i$ get Clopper--Pearson limits. This uses $M=2|E_{\rm int}|+|E_{\rm bdry}|$ statistics; under (T) the per-shot averages over a group replace the single-edge statistics. Throughout, $\delta=10^{-3}$, and $\lambda$ is taken on a grid from $0.05$ to $0.5$.

\item Sycamore, $1$ round. $d=3$ ($8$): certified $0.27$--$0.32$, against observed $0.016$--$0.019$; point estimate $0.12$--$0.13$. $d=5$: infinite, although the point estimate is finite ($0.49$--$0.60$). At $3$ rounds every bound is vacuous or infinite.
\end{itemize}
\emph{The assumptions are testable in part, and they fail in part.} Under (G), detectors of $G_{\rm o}$ that no edge joins are uncorrelated. Paired covariances above the Bonferroni threshold (level $10^{-3}$, normal approximation) occur in $8$ of the $18$ Willow $d=3$ one-round experiments and $6$ of the $8$ at $d=5$. Most are marginal ($|z|\le6$). One pair of final-round detectors recurs with $|z|=15$--$19$ in the $d=5$ and $d=7$ patches, and the $d=7$, $10$-round $Z$ run shows long-range time correlations at one plaquette, typical of leakage. Identical distribution fails as well. In some Sycamore runs, single-detector rates change by factors $2$--$5$ between blocks of $5{,}000$--$10{,}000$ consecutive shots, and the drift statistic over chunks of $1{,}000$ shots exceeds its threshold in $4$ of $8$ Sycamore and $4$ of $18$ Willow one-round $d=3$ runs. The experiments that pass both tests are $7$ Willow $d=3$ runs, with certified $0.065$--$0.11$ against observed $0.0034$--$0.017$, and $4$ Sycamore $d=3$ runs, with $0.27$--$0.31$ against $0.016$--$0.017$.
So the certificate is non-vacuous only at $d\le5$ with one round, and it is conditional on (G). It is $6$--$190$ times the observed rate. At $d=3$ the bound at the point estimate is already $2.7$--$8.5$ times the observed rate, and the confidence limits add a factor $2.2$--$3$. At $d=5$ these factors are $10$--$28$ and $7$--$8$. The limits are widest at the boundary edges, where $q_b$ is determined only by a marginal $a_i$ and inherits the limits of the neighbouring edges. Joint rather than coordinatewise limits for $(a_i,T_{ij})$, a boundary statistic that cancels the neighbouring edges to first order, and a version of (G) with the hyperedges seen in the data would be the next steps.

\paragraph*{Gap refinement.} Theorem~\ref{thm:geodesic} uses minimality of the correction run by run. Using it for all runs of the cycle at once moves the Chernoff factor from the error runs to the distances they bridge, and the computable form can then forbid reversals at the junctions of runs.

\begin{theorem}[Gap refinement]\label{thm:gap}
In the setting of Theorem~\ref{thm:geodesic}, write a failure cycle $\gamma$ of part~(a) cyclically as alternating runs: correction runs $\sigma_1,\dots,\sigma_k$ ($k\ge0$) and error runs $\pi_1,\dots,\pi_k$, where $\pi_j$ goes from the end $z_j$ of $\sigma_j$ to the start $y_{j+1}$ of $\sigma_{j+1}$ (indices mod $k$). Let $\tilde d$ be the distance of $\tilde w$ in $G_{\rm o}$ and $\hat d=\tilde d/s$.
\begin{enumerate}[(a)]
\item (Gap lemma.) $\sum_i\tilde w(\sigma_i)\le\sum_j\tilde d(z_j,y_{j+1})$, and every $\sigma_i$ is a $\tilde w$-geodesic.
\item For $\lambda\ge0$,
\[
\pL^{\rm MWPM}\le\sum_{\gamma\in\mathcal C}\ \sum_{S}\ \prod_{e\in S}q_e\prod_{e\in\gamma\setminus S}(1-q_e)\ \exp\Big(\lambda\Big[\sum_j\hat d(z_j,y_{j+1})-\sum_i\hat w(\sigma_i)\Big]\Big),
\]
the inner sum over the nonempty $S\subseteq\gamma$ whose complement has only geodesic runs (for $S=\gamma$ the exponential is $1$).
\item (Computable form.) Let $\mathcal A$ be the arcs (directed edges) of $G_{\rm o}-b$, $P=(I-B_q)^{-1}$ with $B_q$ the non-backtracking matrix of $\mathcal A$ with entries $q_{e(a')}$, and, for arcs $f,l$,
\[
K_X(f,l)=q_{e(f)}P_{fl}\,e^{\lambda\hat d(\mathrm{tail}f,\mathrm{head}l)},\qquad
K_C(f,l)=c_{e(f)}c_{e(l)}F(\mathrm{head}f,\mathrm{tail}l)\,\mathbf 1\big[\tilde w_{e(f)}+\tilde d(\mathrm{head}f,\mathrm{tail}l)+\tilde w_{e(l)}=\tilde d(\mathrm{tail}f,\mathrm{head}l)\big]
\]
($K_C(f,f)=c_{e(f)}$ if $e(f)$ is a geodesic), with $c_e=(1-q_e)e^{-\lambda\hat w_e}$ and $F(u,w)$ the sum of $\prod c_e$ over the paths from $u$ to $w$ in $G_{\rm o}-b$ that are geodesics of $G_{\rm o}$ ($F(u,u)=1$). Let $J(l,f)=1$ if $\mathrm{head}\,l=\mathrm{tail}\,f$ and $\mathrm{head}\,f\ne\mathrm{tail}\,l$ (no reversal at a junction). If the spectral radius of $M=\begin{pmatrix}0&JK_C\\JK_X&0\end{pmatrix}$ is $<1$ and the walk sums converge, the sum of (b) is at most
\[
T_0+T_1+(s_X,s_C)(I-M)^{-1}\begin{pmatrix}e_X\\e_C\end{pmatrix},
\]
where $T_0$ is the sum of $\prod q_e$ over closed non-backtracking walks $b\to u\to\dots\to v\to b$ with $\sigma(u)=0$, $\sigma(v)=1$ (cycles with $S=\gamma$), and $s_X,s_C,e_X,e_C,T_1$ collect the first and last runs at $b$ as in Theorem~\ref{thm:geodesic}(c) (a first or last run may consist of its boundary edge only; then no junction constraint applies, and the gap through $b$ is bounded by $\hat d(z,b)+\hat d(b,y)$).
\item $\err\le\pL^{\rm MWPM}$, so (b) and (c) bound $\err$ as well.
\end{enumerate}
\hfill\st{PROVED} (draft; checks G4--G6)
\end{theorem}
\begin{proof}
(a) Let $\tau_j$ be a $\tilde w$-geodesic from $z_j$ to $y_{j+1}$. The edge set $\Sigma=\bigoplus_i\sigma_i$ has odd degree exactly at the end points $\{y_i,z_i\}$ (counted with multiplicity mod 2), and so has $T=\bigoplus_j\tau_j$, because the pairs $(z_j,y_{j+1})$ use every end point once. Hence $\Sigma\oplus T$ has even degree everywhere and $C'=C_{\rm o}\oplus\Sigma\oplus T$ has the syndrome of $C_{\rm o}$. As the $\sigma_i\subseteq C_{\rm o}$ are disjoint, $\tilde w(C')\le\tilde w(C_{\rm o})-\sum_i\tilde w(\sigma_i)+\sum_j\tilde w(\tau_j)$, and minimality gives the inequality. The geodesic property is Theorem~\ref{thm:geodesic}(a).
(b) As in Theorem~\ref{thm:geodesic}(b), with the indicator of (a) bounded by $\exp(\lambda[\sum_j\hat d(z_j,y_{j+1})-\sum_i\hat w(\sigma_i)])$; if $k=0$ the cycle lies in $X$ and no condition is used.
(c) Cut $\gamma$ at $b$ and orient it from its end on side $0$. An error run $\pi$ from $z$ to $y$ contributes $\prod_{e\in\pi}q_e\,e^{\lambda\hat d(z,y)}$ and a correction run $\prod c_e$; a gap through $b$ is split at $b$, using $\hat d(z,y)\le\hat d(z,b)+\hat d(b,y)$. An interior error run is a non-backtracking walk, counted in $K_X$ by its first and last arcs; an interior correction run is a geodesic with first arc $f$ and last arc $l$, which is the case iff the three pieces add up to the distance between its end points, counted in $K_C$. Consecutive runs of a simple cycle meet at a vertex and the second does not start with the reverse of the last arc of the first, which is the constraint $J$. Distinct pairs $(\gamma,S)$ give distinct sequences, all terms are nonnegative, and the sum over all sequences is the Neumann series of $M$; convergence is certified as before.
\end{proof}

\begin{proposition}[Gap refinement for every $d$]\label{prop:gap-lattice}
In the circuit-level setting of Proposition~\ref{prop:circuit-lattice} (stim's rotated memory circuit, uniform noise of strength $p$, decoder pymatching), assume (H1)--(H2) and, in reduced coordinates $(x/2,y/2,t)$ with the twelve offsets $v$:
\begin{enumerate}[(A1)]
\item every interior edge with offset $v$ has $q_e\le\hat q(v)$ and $\hat w^-(v)\le\hat w_e\le\hat w^+(v)$, and every boundary edge has $q_e\le\hat q_b$, $\hat w_b^-\le\hat w_e\le\hat w_b^+$, with envelopes that do not depend on $d$;
\item $\hat d(z,y)\le D^+(y-z)$ for all detectors $z,y$, where $\hat d$ is the distance of $\hat w$ in $G_{\rm o}$ and $D^+$ the distance of the weights $\hat w^+$ on the infinite lattice.
\end{enumerate}
For $\lambda,\mu,\theta\ge0$ and $L\ge1$ let $K_X,K_C$ be the $12\times12$ matrices (first offset, last offset) of the lattice sums
\[
\sum_{\text{NB walks}}\prod_{\rm steps}\hat q(v)e^{-\theta\Delta y(v)}\cdot e^{\lambda D^+(\mathrm{end})},\qquad
\sum_{\text{NB walks}}\prod_{\rm steps}e^{-(\lambda+\mu)\hat w^-(v)-\theta\Delta y(v)}\cdot e^{\mu D^+(\mathrm{end})},
\]
evaluated exactly up to length $L$ and beyond it by blocks of length $L$ (row-sum norm $<1$ required). Let $J$ forbid reversals at junctions and $M=\begin{pmatrix}0&JK_C\\JK_X&0\end{pmatrix}$. If $\rho(M)<1$, then for every odd $d\ge5$
\[
\err_d\le\pL^{\rm MWPM}(d)\le\frac{(d+1)^2}{2}\,(x_b+c_b)^2\,F\,e^{-\theta(d-2)},\qquad x_b=\hat q_be^{\lambda\hat w_b^+},\ c_b=e^{-\lambda\hat w_b^-},\ F=1+v_0(I-M)^{-1}\mathbf 1,
\]
with $v_0=(\mathbf 1^{\mathsf T}K_X,\mathbf 1^{\mathsf T}K_C)$. Hence $\liminf_d-\tfrac1d\ln\pL^{\rm MWPM}(d)\ge\theta^\ast=\sup\{\theta:\rho(M)<1\}$.
\hfill\st{PROVED} (draft) for $5\le d\le15$, where (A1), (A2), (H1), (H2) are checked; for $d>15$ conditional on them (computer-assisted, floating point)
\end{proposition}
\begin{proof}
Start from Theorem~\ref{thm:gap}(b) and the run decomposition of its proof (c), oriented from the row $y=2d-2$ to the row $y=2$ as in Proposition~\ref{prop:circuit-lattice}. Map every interior run to its sequence of lattice offsets; non-backtracking walks map to non-backtracking lattice walks and consecutive runs do not reverse at their junction.
An error run $\pi$ from $z$ to $y$ costs $\prod q_e\,e^{\lambda\hat d(z,y)}\le\prod\hat q(v)\,e^{\lambda D^+(y-z)}$ by (A1)--(A2).
A correction run $\sigma$ from $u$ to $v$ is a geodesic of $G_{\rm o}$, so $\sum\hat w^-\le\hat w(\sigma)=\hat d(u,v)\le D^+(v-u)$; its factor $\prod(1-q_e)e^{-\lambda\hat w_e}$ is at most $\prod e^{-\lambda\hat w^-(v)}\cdot\mathbf 1[\sum\hat w^-\le D^+]\le\prod e^{-(\lambda+\mu)\hat w^-(v)}e^{\mu D^+(v-u)}$.
Walks longer than $L$: $D^+$ is a distance, hence subadditive, so $e^{\lambda D^+}$ (and $e^{\mu D^+}$) of a walk is at most the product over its blocks of length $L$ and a remainder; summing over blocks without the junction constraint between them gives the blocked sums. The steps' $\Delta y$ add up to $-(2d-4)$, so the tilt contributes exactly $e^{\theta(d-2)}$, which is compensated by the factor $e^{-\theta(d-2)}$.
The first and last runs contain a boundary edge: an error run starting at $b$ through $u$ costs $q_b\,e^{\lambda\hat d(b,y)}\le\hat q_be^{\lambda\hat w_b^+}e^{\lambda D^+(y-u)}$ (triangle inequality), a correction run $c_b$ times a geodesic continuation; this gives the factors $(x_b+c_b)$ at both ends. Summing the alternating sequences of interior runs (possibly none) with either first type gives at most $F$; there are $(d+1)^2/2$ start detectors by (H2).
\end{proof}
