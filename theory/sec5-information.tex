\section{信息论界与实验设计}\label{sec:info}

\subsection{$\alpha$ 的夹逼:Chernoff 上界与联合下界}

\begin{proposition}[Chernoff 上界]\label{prop:genie}
设 $d\ge3$ 为奇数,考虑旋转表面码族,噪声 $\Nn_{\rm cc}(p,e)$。则
\[
\alpha_+(p,e)\;\le\;\ln\frac1{B(p,e)},\qquad B(p,e)=e+(1-e)\Big(2\sqrt{\tfrac{p(1-p)}3}+\tfrac{2p}3\Big).
\]
更精确地,对每个 $d$ 有 $\err_d\ge\varepsilon_{\rm genie}(d)$,其中
$\varepsilon_{\rm genie}(d)=\tfrac12\bigl(1-\mathrm{TV}(P_1^{\otimes d},Q_1^{\otimes d})\bigr)$,
$P_1$ 是单比特 $(a,f)$ 的联合分布,$Q_1=P_1(\cdot+X)$ 是平移后的分布。\hfill\st{SKETCH}
\end{proposition}
\begin{proof}
取一列内部的 $d$ 个比特 $T$(列号 $c_0$,$1\le c_0\le d-2$),$\bar X_T$ 为这一列上的 $X$ 型逻辑算符。

\emph{步骤 1:$\ker\sigma\cap\mathcal P_T=\{I,\bar X_T\}$。}设 $(x,z)$ 支撑于 $T$ 且综合征为零。每个触及第 $c_0$ 列的体 $Z$ 型方格与这一列恰交于
两个竖直相邻的比特 $(r',r'+1)$,且 $r'=0,\dots,d-2$ 各出现一次(第 $c_0$ 列与第 $c_0-1$ 列的奇偶互补,
内部列两侧方格都存在);故 $x_{r'}=x_{r'+1}$,$x\in\{0,\mathbf 1_T\}$。对 $X$ 型方格同理得 $z$ 为常数;
又顶边的 $X$ 型边界检测器与该列恰交于一个比特 $(0,c_0)$,强制 $z_0=0$,故 $z=0$。(小码 $d=3$ 已由枚举验证。)

\emph{步骤 2:先知。}先知知道 $F$、$E_{T^c}$ 与 $\sigma(E)$,等价于知道 $\sigma(E_T)$。由步骤 1,$E_T$ 只可能是 $a$ 或 $a+\bar X_T$,
二者标签不同。先知的 Bayes 错误率为 $\tfrac12\sum_{a,f_T}\min\big(P_T(a,f_T),P_T(a+\bar X_T,f_T)\big)=\varepsilon_{\rm genie}$。
真实观测 $(\sigma(E),F)$ 是先知观测的粗粒化 (F1),故 $\err_d\ge\varepsilon_{\rm genie}$(定理~\ref{thm:monotone})。

\emph{步骤 3:指数。}$\varepsilon_{\rm genie}$ 是在 $d$ 个 i.i.d.\ 样本上区分 $P_1$ 与 $Q_1$ 的等先验 Bayes 错误率,
其指数为 Chernoff 信息 $C(P_1,Q_1)=-\ln\min_{s\in[0,1]}b_s$ \cite[\S11.9]{CT}。由 $b_s=b_{1-s}$ 的凸性,
最小值在 $s=1/2$,得 $C=-\ln B$。注意标记在联合字母表 $(a,f)$ 中,先知看到它们,所以在 $F=1$ 的位置上两个假设\emph{完全相同},
对 $B$ 贡献 $e\cdot1$,这就是 $B$ 中 $e$ 项的来源。
\end{proof}

\noindent 上述证明依赖标准的 Chernoff 渐近结果(引用),步骤 1 对一般奇数 $d$ 的几何论证已写出;因此标 \st{SKETCH}。
$d=3$ 时 $\err\ge\varepsilon_{\rm genie}$ 已由精确计算验证(检验 C6)。

\begin{proposition}[联合下界]\label{prop:lower}
令 $\Phi(x)=\limsup_{d\to\infty}\tfrac1d\ln\widetilde W_d(x)$(推论~\ref{cor:product})。则
$\alpha_-(p,e)\ge-\Phi\big(B(p,e)\big)$。\hfill\st{PROVED}
\end{proposition}
\begin{proof}
推论~\ref{cor:product}:$\err_d\le\tfrac12\widetilde W_d(B)$,取 $-\tfrac1d\ln$ 并取 $\liminf$。
\end{proof}

\begin{corollary}[夹在两个仅依赖 $B$ 的函数之间]\label{cor:sandwich}
$-\Phi(B(p,e))\le\alpha_-\le\alpha_+\le-\ln B(p,e)$。\hfill\st{PROVED}(对 \ref{prop:genie} 的前提)
\end{corollary}

夹逼并不约束 $\alpha$ 的导数,因此不能由它直接推出汇率的上下界;它的作用是给出可检验的假设 $H_B$
(猜想~\ref{conj:HB}):若 $\alpha$ 在邻域内是 $B$ 的函数,则 $R_\alpha=R_B$。

\subsection{实验设计与检测极限}\label{sec:design}

这一部分对应 P2。记 $T$ 为总观测时长(轮数),事件为速率 $r$ 的泊松过程。

\begin{proposition}[泊松下限]\label{prop:poisson}
设 $H_0$:无突发事件;$H_1$:突发事件为强度 $r$ 的泊松过程(任意形状),其余各层独立且相同。
对任何基于总时长 $T$ 的全部数据的检验 $\varphi$,若 $H_0$ 下第一类错误 $\le a$,则 $H_1$ 下的功效
\[
\Prob_1[\varphi=1]\;\le\;1-e^{-rT}(1-a).
\]
因此要达到功效 $1-\delta$ 必须 $T\ge\ln\!\big((1-a)/\delta\big)/r$。任何利用时空模式的方法都不能突破 $1/r$ 标度。\hfill\st{PROVED}
\end{proposition}
\begin{proof}
$H_1$ 下以概率 $e^{-rT}$ 一次事件都不发生,此时数据与 $H_0$ 下同分布。于是
$\Prob_1[\varphi=1]\le(1-e^{-rT})+e^{-rT}\Prob_0[\varphi=1]\le1-e^{-rT}(1-a)$。
\end{proof}

\begin{proposition}[类别区分的指数链]\label{prop:exponents}
设事件带标记 $a\in\mathcal A$(类别),两个假设的各类强度为 $r_a^{(0)},r_a^{(1)}$。完全观测事件时,
每单位时间的 Stein 指数与 Chernoff 指数为
\[
D=\sum_a\Big[r^{(1)}_a\ln\frac{r^{(1)}_a}{r^{(0)}_a}-r^{(1)}_a+r^{(0)}_a\Big],\qquad
C=\max_{s\in[0,1]}\sum_a\Big[s\,r^{(1)}_a+(1-s)\,r^{(0)}_a-(r^{(1)}_a)^s(r^{(0)}_a)^{1-s}\Big]
\]
(泊松过程的相对熵与 Chernoff 信息)。设实际数据 $Z$ 是事件过程经检测器信道 $W$ 的输出,则
\[
D_{\rm count}\ \le\ D_{\rm pattern}\ \le\ D_{\rm event}=D,\qquad C_{\rm count}\le C_{\rm pattern}\le C,
\]
其中 count 指只用被检测到的事件总数,pattern 指用全部时空模式的最优检验。因此达到同一误差所需轮数之比
$T_{\rm count}/T_{\rm pattern}\to D_{\rm pattern}/D_{\rm count}\ (\delta\to0)$,并以
$D_{\rm event}/D_{\rm count}$ 为上限。\hfill 不等式链 \st{PROVED}(相对熵/R\'enyi 散度的数据处理不等式与泊松的闭式);轮数之比的渐近形式依赖 Stein 引理 \st{SKETCH}
\end{proposition}

\begin{problem}[O4:实验设计定理]\label{prob:O4}
在预算约束下选择码距 $d$、轮数、(自适应)策略,使 $D_{\rm pattern}(d,\text{策略})$ 或 Chernoff 指数最大;
并刻画 $D_{\rm pattern}/D_{\rm count}$ 对检测器信道 $W$ 的依赖。\hfill\st{PLAN}
\end{problem}
\noindent P2 的数值分别估计 $D_{\rm count}$ 与 $D_{\rm pattern}$(Monte Carlo 或解析),报告 $T_{\rm count}/T_{\rm pattern}$ 并对照命题~\ref{prop:poisson}
(任何方法的轮数下限 $\ln(1/\delta)/r$)。
